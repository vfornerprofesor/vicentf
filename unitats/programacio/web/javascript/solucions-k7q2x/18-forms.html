Exercici 18.1

Exercise 18.1. Make the form of the example work, with the four checks and the red borders.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Sign up</title>
    <style>
        .wrong {
            border: 2px solid red;
            background-color: #ffecec;
        }

        .error {
            color: red;
        }

        .ok {
            color: green;
        }
    </style>
</head>

<body>
    <form id="signupForm">
        <p>
            <label for="name">Name</label>
            <input type="text" id="name">
        </p>
        <p>
            <label for="email">Email</label>
            <input type="text" id="email">
        </p>
        <p>
            <label for="age">Age</label>
            <input type="text" id="age">
        </p>
        <p>
            <label for="password">Password</label>
            <input type="password" id="password">
        </p>
        <button type="submit">Sign up</button>
    </form>

    <div id="messages"></div>

    <script>
        function clearMarks() {
            for (const input of document.querySelectorAll("input")) {
                input.classList.remove("wrong");
            }
        }

        function mark(id) {
            document.querySelector(id).classList.add("wrong");
        }

        const form = document.querySelector("#signupForm");
        const messages = document.querySelector("#messages");

        form.addEventListener("submit", function (event) {
            event.preventDefault();
            clearMarks();

            const name = document.querySelector("#name").value.trim();
            const email = document.querySelector("#email").value.trim();
            const age = Number(document.querySelector("#age").value);
            const password = document.querySelector("#password").value;

            const errors = [];

            if (name.length < 3) {
                errors.push("The name must have at least 3 letters.");
                mark("#name");
            }

            if (!email.includes("@") || !email.includes(".")) {
                errors.push("The email must contain an @ and a dot.");
                mark("#email");
            }

            if (isNaN(age) || age < 16 || age > 99) {
                errors.push("The age must be a number between 16 and 99.");
                mark("#age");
            }

            if (password.length < 8) {
                errors.push("The password must have at least 8 characters.");
                mark("#password");
            }

            messages.innerHTML = "";

            if (errors.length > 0) {
                const list = document.createElement("ul");

                for (const error of errors) {
                    const li = document.createElement("li");
                    li.textContent = error;
                    list.appendChild(li);
                }

                messages.appendChild(list);
                messages.className = "error";
            } else {
                messages.textContent = `Welcome, ${name}! Your account is ready.`;
                messages.className = "ok";
                form.reset();       // empties all the fields
            }
        });
    </script>
</body>

</html>
Les quatre comprovacions (nom, email, edat i contrasenya) afigixen una classe wrong (vora roja) al camp i un missatge a la llista d'errors. Amb event.preventDefault() es para l'enviament normal del formulari.

Exercici 18.3

Exercise 18.3. Validate as the user types, not only when they send: use the input event to remove the red border as soon as the field becomes correct. Use a regular expression for the email.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Sign up</title>
    <style>
        .wrong {
            border: 2px solid red;
            background-color: #ffecec;
        }

        .error {
            color: red;
        }

        .ok {
            color: green;
        }
    </style>
</head>

<body>
    <form id="signupForm">
        <p>
            <label for="name">Name</label>
            <input type="text" id="name">
        </p>
        <p>
            <label for="email">Email</label>
            <input type="text" id="email">
        </p>
        <p>
            <label for="age">Age</label>
            <input type="text" id="age">
        </p>
        <p>
            <label for="password">Password</label>
            <input type="password" id="password">
        </p>
        <button type="submit">Sign up</button>
    </form>

    <div id="messages"></div>

    <script>
        function clearMarks() {
            for (const input of document.querySelectorAll("input")) {
                input.classList.remove("wrong");
            }
        }

        function mark(id) {
            document.querySelector(id).classList.add("wrong");
        }

        const form = document.querySelector("#signupForm");
        const messages = document.querySelector("#messages");

        form.addEventListener("submit", function (event) {
            event.preventDefault();
            clearMarks();

            const name = document.querySelector("#name").value.trim();
            const email = document.querySelector("#email").value.trim();
            const age = Number(document.querySelector("#age").value);
            const password = document.querySelector("#password").value;

            const errors = [];

            if (name.length < 3) {
                errors.push("The name must have at least 3 letters.");
                mark("#name");
            }

            if (!email.includes("@") || !email.includes(".")) {
                errors.push("The email must contain an @ and a dot.");
                mark("#email");
            }

            if (isNaN(age) || age < 16 || age > 99) {
                errors.push("The age must be a number between 16 and 99.");
                mark("#age");
            }

            if (password.length < 8) {
                errors.push("The password must have at least 8 characters.");
                mark("#password");
            }

            messages.innerHTML = "";

            if (errors.length > 0) {
                const list = document.createElement("ul");

                for (const error of errors) {
                    const li = document.createElement("li");
                    li.textContent = error;
                    list.appendChild(li);
                }

                messages.appendChild(list);
                messages.className = "error";
            } else {
                messages.textContent = `Welcome, ${name}! Your account is ready.`;
                messages.className = "ok";
                form.reset();       // empties all the fields
            }
        });

        // ---- live validation: the red border disappears as soon as the field is correct ----
        const rules = {
            "#name": value => value.trim().length >= 3,
            "#email": value => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim()),
            "#age": value => value.trim() !== "" && Number(value) >= 16 && Number(value) <= 99,
            "#password": value => value.length >= 8
        };

        for (const selector in rules) {
            const field = document.querySelector(selector);
            field.addEventListener("input", function () {
                field.classList.toggle("wrong", !rules[selector](field.value));
            });
        }
    </script>
</body>

</html>
L'esdeveniment input es dispara a cada tecla: classList.toggle("wrong", condició) posa la vora roja mentre el camp és incorrecte i la lleva quan passa a ser correcte. L'email es valida amb una expressió regular.

Exercici 18.4

Exercise 18.4. Login. User and password. If the user is admin and the password is smx2027, show "Welcome". If not, show "Wrong user or password" and count the failed attempts. After 3, disable the button (button.disabled = true).

Fitxer de partida: ex4.html

Solució

script.js 
const form = document.querySelector("#loginForm");
const button = form.querySelector("button");
const messages = document.querySelector("#messages");
let attempts = 0;

form.addEventListener("submit", function (event) {
    event.preventDefault();

    const user = document.querySelector("#user").value.trim();
    const pass = document.querySelector("#pass").value;

    if (user === "admin" && pass === "smx2027") {
        messages.textContent = "Welcome!";
        messages.className = "ok";
        return;
    }

    attempts++;
    messages.textContent = `Wrong user or password. Attempt ${attempts} of 3.`;
    messages.className = "error";

    if (attempts >= 3) {
        button.disabled = true;
        messages.textContent = "Too many attempts. The form is blocked.";
    }
});

Fitxer de partida: exercises/18-forms/ex4.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Aquest login només és un exercici: mai es posa una contrasenya real en JavaScript, perquè qualsevol persona pot llegir-la amb F12.

Exercici 18.5

Exercise 18.5. Sign-up with a summary. When all the data is correct, do not just say "welcome": draw a card with everything the user typed, and a "Change data" button that brings the form back.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Sign up with summary</title>
    <style>
        .wrong {
            border: 2px solid red;
            background-color: #ffecec;
        }

        .error {
            color: red;
        }

        .ok {
            color: green;
        }
    </style>
    <style>
        .card {
            display: inline-block;
            padding: 16px 24px;
            border: 2px solid green;
            border-radius: 10px;
        }
    </style>
</head>

<body>
    <form id="signupForm">
        <p>
            <label for="name">Name</label>
            <input type="text" id="name">
        </p>
        <p>
            <label for="email">Email</label>
            <input type="text" id="email">
        </p>
        <p>
            <label for="age">Age</label>
            <input type="text" id="age">
        </p>
        <p>
            <label for="password">Password</label>
            <input type="password" id="password">
        </p>
        <button type="submit">Sign up</button>
    </form>

    <div id="messages"></div>

    <div id="summary"></div>

    <script>
        const form = document.querySelector("#signupForm");
        const messages = document.querySelector("#messages");
        const summary = document.querySelector("#summary");

        function mark(id, wrong) {
            document.querySelector(id).classList.toggle("wrong", wrong);
        }

        form.addEventListener("submit", function (event) {
            event.preventDefault();

            const name = document.querySelector("#name").value.trim();
            const email = document.querySelector("#email").value.trim();
            const age = Number(document.querySelector("#age").value);
            const password = document.querySelector("#password").value;

            const wrongName = name.length < 3;
            const wrongEmail = !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
            const wrongAge = isNaN(age) || age < 16 || age > 99;
            const wrongPassword = password.length < 8;

            mark("#name", wrongName);
            mark("#email", wrongEmail);
            mark("#age", wrongAge);
            mark("#password", wrongPassword);

            if (wrongName || wrongEmail || wrongAge || wrongPassword) {
                messages.textContent = "Please, correct the fields in red.";
                messages.className = "error";
                return;
            }

            // everything is correct: hide the form and draw a card with the data
            messages.textContent = "";
            form.style.display = "none";

            summary.innerHTML = "";
            const card = document.createElement("div");
            card.className = "card";

            const title = document.createElement("h2");
            title.textContent = "Welcome, " + name + "!";
            card.appendChild(title);

            const data = { Name: name, Email: email, Age: age };
            for (const label in data) {
                const p = document.createElement("p");
                p.textContent = label + ": " + data[label];
                card.appendChild(p);
            }

            const change = document.createElement("button");
            change.id = "change";
            change.textContent = "Change data";
            change.addEventListener("click", function () {
                summary.innerHTML = "";
                form.style.display = "";           // the form comes back, with its data
            });
            card.appendChild(change);

            summary.appendChild(card);
        });
    </script>
</body>

</html>
Si tot és correcte s'amaga el formulari i es dibuixa una targeta amb les dades (creada amb createElement); el botó "Change data" torna a mostrar el formulari amb les dades que hi havia.
