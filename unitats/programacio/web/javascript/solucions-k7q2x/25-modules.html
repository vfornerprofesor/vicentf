Exercici 25.1

Exercise 25.1. Move Product and BankAccount to js/model.js with export, and import them in js/main.js. The page must work exactly as before.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Shop and bank</title>
</head>

<body>
    <h1>Products</h1>
    <ul id="products"></ul>

    <h1>Bank account</h1>
    <p id="account"></p>
    <input type="number" id="amount" placeholder="Amount">
    <button id="deposit">Deposit</button>
    <p id="message"></p>

    <script type="module" src="js/main.js"></script>
</body>

</html>
js/model.js 
/** A product of the shop. */
export class Product {
    #price;

    constructor(name, price, stock = 0) {
        this.name = name;
        this.price = price;
        this.stock = stock;
    }

    get price() {
        return this.#price;
    }

    set price(value) {
        if (value < 0) {
            throw new Error("The price cannot be negative");
        }
        this.#price = value;
    }

    /** Value of the stock of this product. */
    stockValue() {
        return this.price * this.stock;
    }

    toString() {
        return `${this.name}: ${this.price} euros, ${this.stock} in stock`;
    }
}

/** A bank account. */
export class BankAccount {
    #balance = 0;

    constructor(owner, balance = 0) {
        this.owner = owner;
        this.#balance = balance;
    }

    get balance() {
        return this.#balance;
    }

    deposit(amount) {
        if (amount <= 0) {
            throw new Error("The amount must be positive");
        }
        this.#balance += amount;
    }

    withdraw(amount) {
        if (amount > this.#balance) {
            throw new Error("There is not enough money in the account");
        }
        this.#balance -= amount;
    }

    toString() {
        return `${this.owner}: ${this.#balance.toFixed(2)} euros`;
    }
}
js/main.js 
import { Product, BankAccount } from "./model.js";

const products = [
    new Product("Keyboard", 25, 10),
    new Product("Monitor", 120, 3)
];
const account = new BankAccount("Ana", 100);

// Draws the products as a list
function drawProducts() {
    const list = document.querySelector("#products");
    for (const p of products) {
        const li = document.createElement("li");
        li.textContent = p.toString();
        list.appendChild(li);
    }
}

// Draws the account and the message
function drawAccount() {
    document.querySelector("#account").textContent = account.toString();
}

document.querySelector("#deposit").addEventListener("click", function () {
    const message = document.querySelector("#message");
    try {
        account.deposit(Number(document.querySelector("#amount").value));
        message.textContent = "";
    } catch (error) {
        message.textContent = error.message;
    }
    drawAccount();
});

drawProducts();
drawAccount();
Les classes viuen en js/model.js amb export i es porten a js/main.js amb import { Product, BankAccount } from "./model.js". Cal type="module" en el <script> i obrir la pàgina amb un servidor (Live Server); amb file:// els mòduls no funcionen.

js/tests.js 
import { Product, BankAccount } from "./model.js";

// ---- Product ----
const mouse = new Product("Mouse", 15, 4);
console.assert(mouse.stockValue() === 60, "stockValue: 15 x 4 = 60");
console.assert(new Product("Cable", 5).stockValue() === 0, "a product with no stock is worth 0");
console.assert(mouse.toString() === "Mouse: 15 euros, 4 in stock", "toString of Product");

let error = null;
try {
    new Product("Bad", -1);
} catch (e) {
    error = e;
}
console.assert(error !== null && error.message === "The price cannot be negative", "a negative price throws an error");

// ---- BankAccount ----
const account = new BankAccount("Ana", 100);
account.deposit(50);
console.assert(account.balance === 150, "deposit adds money");
account.withdraw(30);
console.assert(account.balance === 120, "withdraw takes money out");

error = null;
try {
    account.withdraw(1000);
} catch (e) {
    error = e;
}
console.assert(error !== null, "withdraw throws an error if there is not enough money");
console.assert(account.toString() === "Ana: 120.00 euros", "toString of BankAccount");

console.log("Tests finished");
Els tests importen les mateixes classes que la pàgina i comproven els seus mètodes amb console.assert (en són huit). Si tot passa, la consola només mostra "Tests finished"; si una afirmació falla, mostra "Assertion failed" amb el missatge.

js/table.js 
import { Product } from "./model.js";     // the same model.js as the first page

const products = [
    new Product("Keyboard", 25, 10),
    new Product("Mouse", 15, 0),
    new Product("Monitor", 120, 3)
];

const table = document.querySelector("#table");
table.innerHTML = "<tr><th>Name</th><th>Price</th><th>Stock</th><th>Value</th></tr>";

for (const p of products) {
    const tr = document.createElement("tr");

    for (const text of [p.name, p.price, p.stock, p.stockValue()]) {
        const td = document.createElement("td");
        td.textContent = text;
        tr.appendChild(td);
    }

    table.appendChild(tr);
}
