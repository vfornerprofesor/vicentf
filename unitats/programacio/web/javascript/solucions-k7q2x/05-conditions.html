Exercici 5.1

Exercise 5.1. Traffic light. Three buttons: Stop, Wait, Go. Each one paints the circle in the page with the right colour and writes the message under it.

Fitxer de partida: ex1.html

Solució

script.js 
const circle = document.querySelector("#circle");
const message = document.querySelector("#message");

function show(colour, text) {
    circle.style.backgroundColor = colour;
    message.textContent = text;
}

document.querySelector("#stopButton").addEventListener("click", function () {
    show("red", "Stop! Do not cross.");
});

document.querySelector("#waitButton").addEventListener("click", function () {
    show("orange", "Wait. Get ready.");
});

document.querySelector("#goButton").addEventListener("click", function () {
    show("green", "Go! You can cross.");
});

Fitxer de partida: exercises/05-conditions/ex1.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Exercici 5.2

Exercise 5.2. Age check. The user types their age. Show "Child", "Teenager", "Adult" or "Senior", each one in a different colour. If the number is negative or over 120, show an error message in red.

Fitxer de partida: ex2.html

Solució

script.js 
const input = document.querySelector("#age");
const result = document.querySelector("#result");

document.querySelector("#checkButton").addEventListener("click", function () {
    const age = Number(input.value);

    if (input.value === "" || age < 0 || age > 120) {
        result.textContent = "Error: type an age between 0 and 120.";
        result.style.color = "red";
    } else if (age < 13) {
        result.textContent = "Child";
        result.style.color = "blue";
    } else if (age < 20) {
        result.textContent = "Teenager";
        result.style.color = "orange";
    } else if (age < 65) {
        result.textContent = "Adult";
        result.style.color = "green";
    } else {
        result.textContent = "Senior";
        result.style.color = "purple";
    }
});

Fitxer de partida: exercises/05-conditions/ex2.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

L'ordre de les condicions importa: primer es descarten els valors impossibles i després es va de menor a major edat, de manera que cada else if ja sap que l'anterior era falsa.

Resultat

Exercici 5.3

Exercise 5.3. Password check. The user types a password. Show, one under the other, whether it has at least 8 characters, whether it has a number, and whether it has a capital letter. Green tick or red cross for each one.

Fitxer de partida: ex3.html

Solució

script.js 
const input = document.querySelector("#password");
const result = document.querySelector("#result");

// Adds one line to the result: green tick or red cross
function addLine(ok, text) {
    const line = document.createElement("p");
    line.className = ok ? "ok" : "ko";
    line.textContent = (ok ? "\u2714 " : "\u2718 ") + text;
    line.style.color = ok ? "green" : "red";
    result.appendChild(line);
}

document.querySelector("#checkButton").addEventListener("click", function () {
    const password = input.value;
    result.textContent = "";   // clear the previous result

    addLine(password.length >= 8, "At least 8 characters");
    addLine(/[0-9]/.test(password), "Has a number");
    addLine(/[A-Z]/.test(password), "Has a capital letter");
});

Fitxer de partida: exercises/05-conditions/ex3.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Les expressions regulars /[0-9]/ i /[A-Z]/ comproven si hi ha algun número o alguna majúscula (es veuen millor en la lliçó 13). També es pot fer amb un bucle i "0123456789".includes(letter).

Exercici 5.4

Exercise 5.4. Leap year. The user types a year and the page says if it is a leap year. (A leap year is divisible by 4, but not by 100, unless it is also divisible by 400.)

Fitxer de partida: ex4.html

Solució

script.js 
const input = document.querySelector("#year");
const result = document.querySelector("#result");

document.querySelector("#checkButton").addEventListener("click", function () {
    const year = Number(input.value);

    if ((year % 4 === 0 && year % 100 !== 0) || year % 400 === 0) {
        result.textContent = `${year} is a leap year.`;
        result.style.color = "green";
    } else {
        result.textContent = `${year} is NOT a leap year.`;
        result.style.color = "red";
    }
});

Fitxer de partida: exercises/05-conditions/ex4.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Un any és de traspàs si és divisible per 4 i no per 100, o si és divisible per 400. 2024 sí, 1900 no, 2000 sí.

Resultat

Exercici 5.5

Exercise 5.5. Calculator, first version. Two numbers, a <select> with + - * /, and a button. Show the result. If the user divides by zero, show "You cannot divide by zero" in red.

Fitxer de partida: ex5.html

Solució

script.js 
const first = document.querySelector("#firstNumber");
const second = document.querySelector("#secondNumber");
const operation = document.querySelector("#operation");
const result = document.querySelector("#result");

document.querySelector("#calculateButton").addEventListener("click", function () {
    const a = Number(first.value);
    const b = Number(second.value);
    result.style.color = "";

    if (operation.value === "/" && b === 0) {
        result.textContent = "You cannot divide by zero";
        result.style.color = "red";
    } else if (operation.value === "+") {
        result.textContent = a + b;
    } else if (operation.value === "-") {
        result.textContent = a - b;
    } else if (operation.value === "*") {
        result.textContent = a * b;
    } else {
        result.textContent = a / b;
    }
});

Fitxer de partida: exercises/05-conditions/ex5.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Exercici 5.6

Exercise 5.6. Cinema ticket. A ticket costs 8 €. Under 14 and over 65 pay 5 €. On Wednesday everybody pays 5 €. The user types the age and chooses the day; the page shows the price.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Cinema ticket</title>
</head>

<body>
    <h1>Cinema ticket</h1>
    <label for="age">Age</label>
    <input type="number" id="age" min="0">
    <label for="day">Day</label>
    <select id="day">
        <option value="monday">Monday</option>
        <option value="tuesday">Tuesday</option>
        <option value="wednesday">Wednesday</option>
        <option value="thursday">Thursday</option>
        <option value="friday">Friday</option>
        <option value="saturday">Saturday</option>
        <option value="sunday">Sunday</option>
    </select>
    <button id="price">Price</button>
    <p id="output"></p>

    <script>
        document.querySelector("#price").addEventListener("click", function () {
            const age = Number(document.querySelector("#age").value);
            const day = document.querySelector("#day").value;
            let price = 8;

            if (day === "wednesday" || age < 14 || age > 65) {
                price = 5;
            }

            document.querySelector("#output").textContent = `Your ticket costs ${price} euros.`;
        });
    </script>
</body>

</html>

Exercici 5.7

Exercise 5.7. Final mark. Three inputs with three marks. Show the average and, next to it, the word (Fail, Pass, Good, Excellent) with the right colour.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Final mark</title>
</head>

<body>
    <h1>Final mark</h1>
    <input type="number" id="mark1" placeholder="Mark 1" min="0" max="10">
    <input type="number" id="mark2" placeholder="Mark 2" min="0" max="10">
    <input type="number" id="mark3" placeholder="Mark 3" min="0" max="10">
    <button id="calculate">Calculate</button>
    <p id="output"></p>

    <script>
        document.querySelector("#calculate").addEventListener("click", function () {
            const m1 = Number(document.querySelector("#mark1").value);
            const m2 = Number(document.querySelector("#mark2").value);
            const m3 = Number(document.querySelector("#mark3").value);
            const average = (m1 + m2 + m3) / 3;

            let word;
            let colour;
            if (average < 5) {
                word = "Fail";
                colour = "red";
            } else if (average < 7) {
                word = "Pass";
                colour = "orange";
            } else if (average < 9) {
                word = "Good";
                colour = "blue";
            } else {
                word = "Excellent";
                colour = "green";
            }

            const output = document.querySelector("#output");
            output.textContent = `Average: ${average.toFixed(2)} - ${word}`;
            output.style.color = colour;
        });
    </script>
</body>

</html>

Extra · ex6 (contact form)

Contact form (fitxer exercises/05-conditions/ex6.html, que no està enllaçat en cap lliçó). Check that the email and password fields are not empty, using &&. Show the success or the error message in #validationMessage.

Solució

script.js 
const email = document.querySelector("#email");
const password = document.querySelector("#password");
const message = document.querySelector("#validationMessage");

document.querySelector("#sendButton").addEventListener("click", function () {
    if (email.value !== "" && password.value !== "") {
        message.textContent = "Data sent correctly.";
        message.style.color = "green";
    } else {
        message.textContent = "Error: both fields are required.";
        message.style.color = "red";
    }
});
