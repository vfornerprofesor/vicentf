Exercici 19.1

Exercise 19.1. Ask the user's name the first time and save it. The next time the page loads, greet them directly, with no question. Add a "Forget me" button.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Hello again</title>
</head>

<body>
    <h1 id="output">Hello!</h1>
    <input type="text" id="nameInput" placeholder="Your name">
    <button id="saveButton">Save</button>
    <button id="forgetButton">Forget me</button>

    <script>
        const output = document.querySelector("#output");
        const nameInput = document.querySelector("#nameInput");
        const saveButton = document.querySelector("#saveButton");
        const forgetButton = document.querySelector("#forgetButton");

        const name = localStorage.getItem("username");

        if (name === null) {
            output.textContent = "Hello! What is your name?";
        } else {
            output.textContent = `Welcome back, ${name}!`;
        }

        saveButton.addEventListener("click", function () {
            const value = nameInput.value.trim();

            if (value === "") {
                return;
            }

            localStorage.setItem("username", value);
            output.textContent = `Nice to meet you, ${value}!`;
        });

        forgetButton.addEventListener("click", function () {
            localStorage.removeItem("username");
            output.textContent = "I have forgotten you. Reload the page.";
        });
    </script>
</body>

</html>
En la previsualització el localStorage està simulat en memòria, així que no s'ensenya la persistència en recarregar; en un fitxer real les dades queden guardades en el navegador.

Exercici 19.2

Exercise 19.2. A click counter that survives a reload.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Click counter</title>
</head>

<body>
    <h1>Click counter</h1>
    <button id="click">Click me</button>
    <p id="count"></p>

    <script>
        let clicks = Number(localStorage.getItem("clicks")) || 0;     // 0 the first time
        const count = document.querySelector("#count");

        function show() {
            count.textContent = `Clicks: ${clicks}`;
        }

        document.querySelector("#click").addEventListener("click", function () {
            clicks++;
            localStorage.setItem("clicks", clicks);
            show();
        });

        show();
    </script>
</body>

</html>
En la previsualització el localStorage està simulat en memòria, així que no s'ensenya la persistència en recarregar; en un fitxer real les dades queden guardades en el navegador.

Exercici 19.3

Exercise 19.3. Add localStorage to your mini-project 2. This is the important one: your list must still be there after F5.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>A list that is saved</title>
</head>

<body>
    <h1>My list</h1>
    <input type="text" id="itemInput" placeholder="Something">
    <button id="addButton">Add</button>
    <ul id="list"></ul>
    <p id="counter"></p>

    <script>
        const input = document.querySelector("#itemInput");
        const list = document.querySelector("#list");
        const counter = document.querySelector("#counter");
        let items = load();

        // Reads the array from localStorage (an empty array if there is nothing or it is broken)
        function load() {
            try {
                return JSON.parse(localStorage.getItem("items")) || [];
            } catch (error) {
                return [];
            }
        }

        // Saves the array
        function save() {
            localStorage.setItem("items", JSON.stringify(items));
        }

        function draw() {
            list.textContent = "";

            items.forEach(function (text, index) {
                const li = document.createElement("li");
                li.textContent = text;

                const remove = document.createElement("button");
                remove.textContent = "Delete";
                remove.addEventListener("click", function () {
                    items.splice(index, 1);
                    save();
                    draw();
                });

                li.appendChild(remove);
                list.appendChild(li);
            });

            counter.textContent = items.length + (items.length === 1 ? " item" : " items");
        }

        document.querySelector("#addButton").addEventListener("click", function () {
            const text = input.value.trim();
            if (text === "") {
                return;
            }
            items.push(text);
            input.value = "";
            save();
            draw();
        });

        draw();
    </script>
</body>

</html>
La clau és la funció load() (llig amb JSON.parse i, si no hi ha res, torna un array buit) i la funció save() (guarda amb JSON.stringify), cridada cada vegada que l'array canvia. En obrir la pàgina es fa let items = load() abans de dibuixar. En la previsualització el localStorage està simulat en memòria, així que no s'ensenya la persistència en recarregar; en un fitxer real les dades queden guardades en el navegador.

Exercici 19.5

Exercise 19.5. Save an array of objects: products with name and price. Draw them as cards and keep them after a reload.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products</title>
    <style>
        .card {
            display: inline-block;
            margin: 6px;
            padding: 10px 16px;
            border: 2px solid #cccccc;
            border-radius: 8px;
        }
    </style>
</head>

<body>
    <h1>Products</h1>
    <input type="text" id="nameInput" placeholder="Name">
    <input type="number" id="priceInput" placeholder="Price">
    <button id="addButton">Add</button>
    <div id="cards"></div>

    <script>
        let products = load();

        function load() {
            try {
                return JSON.parse(localStorage.getItem("products")) || [];
            } catch (error) {
                return [];
            }
        }

        function draw() {
            const cards = document.querySelector("#cards");
            cards.textContent = "";

            for (const p of products) {
                const card = document.createElement("div");
                card.className = "card";

                const title = document.createElement("h3");
                title.textContent = p.name;
                const price = document.createElement("p");
                price.textContent = p.price + " euros";

                card.append(title, price);
                cards.appendChild(card);
            }
        }

        document.querySelector("#addButton").addEventListener("click", function () {
            const name = document.querySelector("#nameInput").value.trim();
            const price = Number(document.querySelector("#priceInput").value);

            if (name === "" || isNaN(price)) {
                return;
            }

            products.push({ name: name, price: price });     // an array of objects
            localStorage.setItem("products", JSON.stringify(products));
            draw();
        });

        draw();
    </script>
</body>

</html>
En la previsualització el localStorage està simulat en memòria, així que no s'ensenya la persistència en recarregar; en un fitxer real les dades queden guardades en el navegador.

Exercici 19.6

Exercise 19.6. Dark mode. A button that adds or removes the class dark on <body>, and remembers the choice in localStorage.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Dark mode</title>
    <style>
        body.dark {
            background-color: #222222;
            color: #eeeeee;
        }
    </style>
</head>

<body>
    <h1>Dark mode</h1>
    <p>This page remembers your choice.</p>
    <button id="toggle">Change mode</button>

    <script>
        // at the beginning: apply the saved choice
        if (localStorage.getItem("dark") === "yes") {
            document.body.classList.add("dark");
        }

        document.querySelector("#toggle").addEventListener("click", function () {
            document.body.classList.toggle("dark");
            const isDark = document.body.classList.contains("dark");
            localStorage.setItem("dark", isDark ? "yes" : "no");
        });
    </script>
</body>

</html>
En la previsualització el localStorage està simulat en memòria, així que no s'ensenya la persistència en recarregar; en un fitxer real les dades queden guardades en el navegador.

Exercici 19.8

Exercise 19.8. In your mini-project 2, show the array with console.table every time it is saved, and in the page the number of items with toLocaleString.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>A list that is saved</title>
</head>

<body>
    <h1>My list</h1>

        const input = document.querySelector("#itemInput");
        const list = document.querySelector("#list");
        const counter = document.querySelector("#counter");
        let items = load();

        // Reads the array from localStorage (an empty array if there is nothing or it is broken)
        function load() {
            try {
                return JSON.parse(localStorage.getItem("items")) || [];
            } catch (error) {
                return [];
            }
        }

        // Saves the array and writes it in the console as a table
        function save() {
            localStorage.setItem("items", JSON.stringify(items));
            console.table(items.map(text => ({ item: text })));
        }

        function draw() {
            list.textContent = "";

            items.forEach(function (text, index) {
                const li = document.createElement("li");
                li.textContent = text;

                const remove = document.createElement("button");

                    items.splice(index, 1);
                    save();
                    draw();
                });

                li.appendChild(remove);
                list.appendChild(li);
            });

            counter.textContent = "Items: " + items.length.toLocaleString("es-ES");
        }

        document.querySelector("#addButton").addEventListener("click", function () {
            const text = input.value.trim();
            if (text === "") {
                return;
            }
            items.push(text);
            input.value = "";
            save();
            draw();
        });

        draw();
    </script>
</body>

</html>
console.table mostra l'array com una taula en la consola cada vegada que es guarda; toLocaleString("es-ES") dona el número amb el format de la llengua (per exemple 1.250 amb punt de milers).

Exercici 19.9

Exercise 19.9. Save the date of the last change and show it under the list: "Last change: 20/5/2027".

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>A list that is saved</title>
</head>

<body>
    <h1>My list</h1>
    <input type="text" id="itemInput" placeholder="Something">
    <button id="addButton">Add</button>
    <ul id="list"></ul>
    <p id="counter"></p>
    <p id="lastChange"></p>

    <script>
        const input = document.querySelector("#itemInput");
        const list = document.querySelector("#list");
        const counter = document.querySelector("#counter");
        let items = load();

        // Reads the array from localStorage (an empty array if there is nothing or it is broken)
        function load() {
            try {
                return JSON.parse(localStorage.getItem("items")) || [];
            } catch (error) {
                return [];
            }
        }

        // Saves the array
        function save() {
            localStorage.setItem("items", JSON.stringify(items));
            localStorage.setItem("lastChange", new Date().toISOString());
        }

        function draw() {
            list.textContent = "";

            items.forEach(function (text, index) {
                const li = document.createElement("li");
                li.textContent = text;

                const remove = document.createElement("button");
                remove.textContent = "Delete";
                remove.addEventListener("click", function () {
                    items.splice(index, 1);
                    save();
                    draw();
                });

                li.appendChild(remove);
                list.appendChild(li);
            });

            counter.textContent = items.length + (items.length === 1 ? " item" : " items");

            const last = localStorage.getItem("lastChange");
            document.querySelector("#lastChange").textContent =
                last ? "Last change: " + new Date(last).toLocaleDateString("es-ES") : "";
        }

        document.querySelector("#addButton").addEventListener("click", function () {
            const text = input.value.trim();
            if (text === "") {
                return;
            }
            items.push(text);
            input.value = "";
            save();
            draw();
        });

        draw();
    </script>
</body>

</html>
La data es guarda amb new Date().toISOString() i es mostra amb toLocaleDateString("es-ES"), que dona el format 20/5/2027.
