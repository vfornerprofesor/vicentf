Exercici 10.1

Exercise 10.1. Create an array with the names of five classmates and draw it as a list in the page. Show the first, the last and the total number under it.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Classmates</title>
</head>

<body>
    <h1>Classmates</h1>
    <ul id="list"></ul>
    <p id="summary"></p>

    <script>
        const classmates = ["Pere", "Joan", "Marta", "Nuria", "Laia"];

        let html = "";
        for (const name of classmates) {
            html += `<li>${name}</li>`;
        }
        document.querySelector("#list").innerHTML = html;

        document.querySelector("#summary").textContent =
            `First: ${classmates[0]} | Last: ${classmates[classmates.length - 1]} | Total: ${classmates.length}`;
    </script>
</body>

</html>

Exercici 10.2

Exercise 10.2. Shopping list. An input, an "Add" button and a list. Every click adds the text to the array and redraws the list. The empty field adds nothing.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Shopping list</title>
</head>

<body>
    <h1>Shopping list</h1>
    <input type="text" id="item" placeholder="Something to buy">
    <button id="add">Add</button>
    <ul id="list"></ul>

    <script>
        const items = [];

        function draw() {
            let html = "";
            for (const item of items) {
                html += `<li>${item}</li>`;
            }
            document.querySelector("#list").innerHTML = html;
        }

        document.querySelector("#add").addEventListener("click", function () {
            const input = document.querySelector("#item");
            const text = input.value.trim();

            if (text !== "") {          // the empty field adds nothing
                items.push(text);
                draw();
            }
            input.value = "";
        });
    </script>
</body>

</html>
Cada clic afig el text a l'array i torna a dibuixar tota la llista. Si el camp està buit, no s'afig res (el segon clic de la prova).

Exercici 10.3

Exercise 10.3. Add a "Remove last" button and a "Clear all" button to the previous exercise.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Shopping list</title>
</head>

<body>
    <h1>Shopping list</h1>
    <input type="text" id="item" placeholder="Something to buy">
    <button id="add">Add</button>
    <button id="removeLast">Remove last</button>
    <button id="clearAll">Clear all</button>
    <ul id="list"></ul>

    <script>
        const items = [];

        function draw() {
            let html = "";
            for (const item of items) {
                html += `<li>${item}</li>`;
            }
            document.querySelector("#list").innerHTML = html;
        }

        document.querySelector("#add").addEventListener("click", function () {
            const input = document.querySelector("#item");
            const text = input.value.trim();
            if (text !== "") {
                items.push(text);
                draw();
            }
            input.value = "";
        });

        document.querySelector("#removeLast").addEventListener("click", function () {
            items.pop();
            draw();
        });

        document.querySelector("#clearAll").addEventListener("click", function () {
            items.length = 0;      // empties the array
            draw();
        });
    </script>
</body>

</html>

Exercici 10.4

Exercise 10.4. With an array of 8 marks: draw them as a list, and under it show the average, how many passed and how many failed. Failed marks appear in red.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Marks</title>
</head>

<body>
    <h1>Marks</h1>
    <ul id="list"></ul>
    <p id="summary"></p>

    <script>
        const marks = [4, 8, 6, 2, 9, 5, 7, 3.5];

        let html = "";
        let sum = 0;
        let passed = 0;
        let failed = 0;

        for (const mark of marks) {
            if (mark >= 5) {
                passed++;
                html += `<li>${mark}</li>`;
            } else {
                failed++;
                html += `<li style="color: red">${mark}</li>`;
            }
            sum += mark;
        }

        document.querySelector("#list").innerHTML = html;
        document.querySelector("#summary").textContent =
            `Average: ${(sum / marks.length).toFixed(2)} | Passed: ${passed} | Failed: ${failed}`;
    </script>
</body>

</html>

Exercici 10.5

Exercise 10.5. Write a function contains(list, value) that returns true or false, without using includes. Use it to tell the user whether the name they typed is in the list.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Is it in the list?</title>
</head>

<body>
    <h1>Is it in the list?</h1>
    <input type="text" id="name" placeholder="A name">
    <button id="check">Check</button>
    <p id="out"></p>

    <script>
        const names = ["Pere", "Joan", "Marta", "Nuria", "Laia"];

        function contains(list, value) {
            for (const item of list) {
                if (item === value) {
                    return true;
                }
            }
            return false;
        }

        document.querySelector("#check").addEventListener("click", function () {
            const name = document.querySelector("#name").value;
            const out = document.querySelector("#out");

            if (contains(names, name)) {
                out.textContent = `${name} is in the list.`;
            } else {
                out.textContent = `${name} is NOT in the list.`;
            }
        });
    </script>
</body>

</html>

Resultat

Exercici 10.6

Exercise 10.6. Write a function noDuplicates(list) that returns a new array with no repeated values. Show the original list and the clean one side by side.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>No duplicates</title>
    <style>
        .columns {
            display: flex;
            gap: 40px;
        }
    </style>
</head>

<body>
    <h1>No duplicates</h1>
    <div class="columns">
        <div>
            <h2>Original</h2>
            <ul id="original"></ul>
        </div>
        <div>
            <h2>Clean</h2>
            <ul id="clean"></ul>
        </div>
    </div>

    <script>
        function noDuplicates(list) {
            const result = [];
            for (const item of list) {
                if (!result.includes(item)) {
                    result.push(item);
                }
            }
            return result;
        }

        function draw(id, list) {
            let html = "";
            for (const item of list) {
                html += `<li>${item}</li>`;
            }
            document.querySelector(id).innerHTML = html;
        }

        const list = ["Pere", "Joan", "Pere", "Marta", "Joan", "Nuria"];
        draw("#original", list);
        draw("#clean", noDuplicates(list));
    </script>
</body>

</html>

Exercici 10.7

Exercise 10.7. A search box: as the user types, show only the names of the array that contain what they typed. (.toLowerCase() and .includes())

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Search</title>
</head>

<body>
    <h1>Search</h1>
    <input type="text" id="search" placeholder="Type to filter...">
    <ul id="list"></ul>

    <script>
        const names = ["Pere", "Joan", "Marta", "Nuria", "Laia", "Jordi", "Maria"];

        function draw(list) {
            let html = "";
            for (const name of list) {
                html += `<li>${name}</li>`;
            }
            document.querySelector("#list").innerHTML = html;
        }

        document.querySelector("#search").addEventListener("input", function () {
            const text = this.value.toLowerCase();
            const filtered = [];

            for (const name of names) {
                if (name.toLowerCase().includes(text)) {
                    filtered.push(name);
                }
            }
            draw(filtered);
        });

        draw(names);   // at the beginning, everyone is shown
    </script>
</body>

</html>
La prova escriu "mar" i queden Marta i Maria. S'usa .toLowerCase() en tots dos costats perquè no importe si hi ha majúscules.

Resultat

Exercici 10.9

Exercise 10.9. When the user clicks on an empty cell, put X or O in turns: change the array and redraw. Show a message when a row is full of the same symbol.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Tic-tac-toe</title>
    <style>
        td {
            width: 60px;
            height: 60px;
            border: 2px solid #333333;
            text-align: center;
            font-size: 2rem;
            cursor: pointer;
        }
    </style>
</head>

<body>
    <h1>Tic-tac-toe</h1>
    <table id="board"></table>
    <p id="message"></p>

    <script>
        const board = [
            ["", "", ""],
            ["", "", ""],
            ["", "", ""]
        ];
        let turn = "X";
        let finished = false;

        function draw() {
            let html = "";
            for (let row = 0; row < 3; row++) {
                html += "<tr>";
                for (let column = 0; column < 3; column++) {
                    html += `<td data-row="${row}" data-col="${column}">${board[row][column]}</td>`;
                }
                html += "</tr>";
            }
            document.querySelector("#board").innerHTML = html;
        }

        // true if a row is full of the same symbol
        function rowFull(row) {
            return board[row][0] !== "" && board[row][0] === board[row][1] && board[row][1] === board[row][2];
        }

        document.querySelector("#board").addEventListener("click", function (event) {
            const cell = event.target.closest("td");
            if (!cell || finished) {
                return;
            }
            const row = Number(cell.dataset.row);
            const column = Number(cell.dataset.col);

            if (board[row][column] !== "") {
                return;                       // the cell is not empty
            }

            board[row][column] = turn;
            draw();

            if (rowFull(row)) {
                document.querySelector("#message").textContent = `${turn} fills a row!`;
                finished = true;
            }
            turn = turn === "X" ? "O" : "X";
        });

        draw();
    </script>
</body>

</html>

Exercici 10.10

Exercise 10.10. Check columns and diagonals too, and announce the winner.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Tic-tac-toe</title>
    <style>
        td {
            width: 60px;
            height: 60px;
            border: 2px solid #333333;
            text-align: center;
            font-size: 2rem;
            cursor: pointer;
        }
    </style>
</head>

<body>
    <h1>Tic-tac-toe</h1>
    <table id="board"></table>
    <p id="message"></p>

    <script>
        const board = [
            ["", "", ""],
            ["", "", ""],
            ["", "", ""]
        ];
        let turn = "X";
        let finished = false;

        function draw() {
            let html = "";
            for (let row = 0; row < 3; row++) {
                html += "<tr>";
                for (let column = 0; column < 3; column++) {
                    html += `<td data-row="${row}" data-col="${column}">${board[row][column]}</td>`;
                }
                html += "</tr>";
            }
            document.querySelector("#board").innerHTML = html;
        }

        // Returns "X", "O" or "" (nobody has won yet)
        function winner() {
            // rows and columns
            for (let i = 0; i < 3; i++) {
                if (board[i][0] !== "" && board[i][0] === board[i][1] && board[i][1] === board[i][2]) {
                    return board[i][0];
                }
                if (board[0][i] !== "" && board[0][i] === board[1][i] && board[1][i] === board[2][i]) {
                    return board[0][i];
                }
            }
            // diagonals
            if (board[1][1] !== "" && board[0][0] === board[1][1] && board[1][1] === board[2][2]) {
                return board[1][1];
            }
            if (board[1][1] !== "" && board[0][2] === board[1][1] && board[1][1] === board[2][0]) {
                return board[1][1];
            }
            return "";
        }

        document.querySelector("#board").addEventListener("click", function (event) {
            const cell = event.target.closest("td");
            if (!cell || finished) {
                return;
            }
            const row = Number(cell.dataset.row);
            const column = Number(cell.dataset.col);

            if (board[row][column] !== "") {
                return;
            }

            board[row][column] = turn;
            draw();

            const result = winner();
            if (result !== "") {
                document.querySelector("#message").textContent = `The winner is ${result}!`;
                finished = true;
            }
            turn = turn === "X" ? "O" : "X";
        });

        draw();
    </script>
</body>

</html>
La prova guanya X per la diagonal (0,0), (1,1) i (2,2).
