Exercici 16.1

Exercise 16.1. A live character counter: as the user types in a field, show "17 / 100 characters", and turn it red over 100.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Counter</title>
</head>

<body>
    <h1>Live counter</h1>
    <textarea id="text" rows="4" cols="40"></textarea>
    <p id="counter">0 / 100 characters</p>

    <script>
        const text = document.querySelector("#text");
        const counter = document.querySelector("#counter");

        text.addEventListener("input", function () {
            const length = text.value.length;
            counter.textContent = `${length} / 100 characters`;
            counter.style.color = length > 100 ? "red" : "";
        });
    </script>
</body>

</html>

Exercici 16.3

Exercise 16.3. Six colour buttons that change the background of the page, built with one loop and data-colour.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Colour buttons</title>
</head>

<body>
    <h1>Choose a colour</h1>
    <div id="buttons"></div>

    <script>
        const colours = ["tomato", "orange", "gold", "mediumseagreen", "dodgerblue", "orchid"];
        const container = document.querySelector("#buttons");

        // one loop creates the six buttons
        for (const colour of colours) {
            const button = document.createElement("button");
            button.textContent = colour;
            button.dataset.colour = colour;            // <button data-colour="tomato">
            container.appendChild(button);
        }

        container.addEventListener("click", function (event) {
            if (event.target.dataset.colour) {
                document.body.style.backgroundColor = event.target.dataset.colour;
            }
        });
    </script>
</body>

</html>
Un sol escoltador en el contenidor (event delegation) serveix per als sis botons: event.target diu quin s'ha clicat i dataset.colour en dona el color.

Exercici 16.4

Exercise 16.4. A field and an "Add" button: every click adds what you typed to the list, and clears the field. Pressing Enter in the field does the same. The empty field adds nothing.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Add items</title>
</head>

<body>
    <h1>My list</h1>
    <input type="text" id="nameInput" placeholder="Something">
    <button id="addButton">Add</button>
    <div id="output"></div>

    <script>
        const input = document.querySelector("#nameInput");
        const button = document.querySelector("#addButton");
        const items = [];

        function draw() {
            let html = "<ul>";
            for (const item of items) {
                html += `<li>${item}</li>`;
            }
            html += "</ul>";
            document.querySelector("#output").innerHTML = html;
        }

        function addItem() {
            const text = input.value.trim();

            if (text === "") {
                return;
            }

            items.push(text);
            input.value = "";
            input.focus();          // the cursor goes back to the field
            draw();
        }

        button.addEventListener("click", addItem);

        input.addEventListener("keydown", function (event) {
            if (event.key === "Enter") {
                addItem();
            }
        });
    </script>
</body>

</html>
La prova afig "Milk" amb Enter, fa un clic amb el camp buit (no afig res) i afig "Bread" amb el botó: queden 2 elements.

Exercici 16.5

Exercise 16.5. A <select> with three cities. When the user chooses one, the page shows a sentence about it (change event).

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Cities</title>
</head>

<body>
    <h1>Choose a city</h1>
    <select id="city">
        <option value="">-- choose one --</option>
        <option value="valencia">Valencia</option>
        <option value="alacant">Alacant</option>
        <option value="castello">Castello</option>
    </select>
    <p id="output"></p>

    <script>
        const sentences = {
            valencia: "Valencia is famous for the paella and the City of Arts and Sciences.",
            alacant: "Alacant has the Santa Barbara castle and a very long beach.",
            castello: "Castello has the Desert de les Palmes and the Magdalena festival."
        };

        document.querySelector("#city").addEventListener("change", function () {
            document.querySelector("#output").textContent = sentences[this.value] || "";
        });
    </script>
</body>

</html>

Exercici 16.6

Exercise 16.6. An image that changes when the mouse is over it and goes back when it leaves.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Image</title>
</head>

<body>
    <h1>Put the mouse over the image</h1>
    <img id="picture" src="data:image/svg+xml,%3Csvg%20xmlns='http://www.w3.org/2000/svg'%20width='200'%20height='120'%3E%3Crect%20width='200'%20height='120'%20fill='%232d5a85'/%3E%3Ctext%20x='100'%20y='70'%20font-size='28'%20fill='white'%20text-anchor='middle'%3EMouse%20out%3C/text%3E%3C/svg%3E" alt="A blue box" width="200" height="120">

    <script>
        const picture = document.querySelector("#picture");
        const normal = picture.src;
        const over = "data:image/svg+xml,%3Csvg%20xmlns='http://www.w3.org/2000/svg'%20width='200'%20height='120'%3E%3Crect%20width='200'%20height='120'%20fill='%23f2a541'/%3E%3Ctext%20x='100'%20y='70'%20font-size='28'%20fill='%231f3c5a'%20text-anchor='middle'%3EMouse%20over%3C/text%3E%3C/svg%3E";

        picture.addEventListener("mouseover", function () {
            picture.src = over;
        });

        picture.addEventListener("mouseout", function () {
            picture.src = normal;
        });
    </script>
</body>

</html>
Dos esdeveniments: mouseover (el ratolí entra) i mouseout (ix). S'hi han posat imatges SVG en línia perquè l'exemple no necessite fitxers.

Exercici 16.7

Exercise 16.7. Live search. An array of 10 names drawn as a list, and a search box. As the user types, only the matching names stay on the screen.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Live search</title>
</head>

<body>
    <h1>Live search</h1>
    <input type="text" id="search" placeholder="Type a name...">
    <ul id="list"></ul>

    <script>
        const names = ["Pere", "Joan", "Marta", "Nuria", "Laia", "Jordi", "Maria", "Pau", "Clara", "Marc"];

        function draw(list) {
            let html = "";
            for (const name of list) {
                html += `<li>${name}</li>`;
            }
            document.querySelector("#list").innerHTML = html;
        }

        document.querySelector("#search").addEventListener("input", function () {
            const text = this.value.toLowerCase();
            draw(names.filter(name => name.toLowerCase().includes(text)));
        });

        draw(names);
    </script>
</body>

</html>
Amb "mar" queden Marta, Maria i Marc.
