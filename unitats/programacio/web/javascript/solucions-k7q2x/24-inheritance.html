Exercici 24.1

Exercise 24.1. Write Vehicle and Car as above, create two cars and show their description.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Cars</title>
</head>

<body>
    <h1>Cars</h1>
    <ul id="list"></ul>

    <script>
        class Vehicle {
            constructor(brand, model) {
                this.brand = brand;
                this.model = model;
            }

            describe() {
                return `${this.brand} ${this.model}`;
            }
        }

        class Car extends Vehicle {
            constructor(brand, model, doors) {
                super(brand, model);         // first: the constructor of Vehicle
                this.doors = doors;
            }

            describe() {
                return `${super.describe()}, ${this.doors} doors`;
            }
        }

        const cars = [
            new Car("Seat", "Ibiza", 5),
            new Car("Mini", "Cooper", 3)
        ];

        for (const car of cars) {
            const li = document.createElement("li");
            li.textContent = car.describe();
            document.querySelector("#list").appendChild(li);
        }
    </script>
</body>

</html>

Exercici 24.2

Exercise 24.2. Add Motorbike extends Vehicle with the engine size, and override describe() using super.describe().

Solució

script.js 
class Vehicle {
    constructor(brand, model) {
        this.brand = brand;
        this.model = model;
    }

    describe() {
        return `${this.brand} ${this.model}`;
    }
}

class Car extends Vehicle {
    constructor(brand, model, doors) {
        super(brand, model);         // first: the constructor of Vehicle
        this.doors = doors;
    }

    describe() {
        return `${super.describe()}, ${this.doors} doors`;
    }
}

class Motorbike extends Vehicle {
    constructor(brand, model, engineSize) {
        super(brand, model);
        this.engineSize = engineSize;
    }

    describe() {
        return `${super.describe()}, ${this.engineSize} cc`;
    }
}

const bike = new Motorbike("Yamaha", "MT-07", 689);
console.log(bike.describe());           // Yamaha MT-07, 689 cc
super.describe() crida al mètode del pare i després la classe filla afig el que li és propi (la cilindrada).

Exercici 24.3

Exercise 24.3. Starting from bank.html, create InsufficientFundsError, throw it from withdraw, and show a different message for it in the catch.

Fitxer de partida: bank.html

Solució

script.js 
// The class of lesson 22, already finished.
class BankAccount {
    constructor(owner, balance) {
        this.owner = owner;
        this.balance = balance;
    }

    // Adds money to the account.
    deposit(amount) {
        this.balance += amount;
    }

    // Takes money out. Throws an InsufficientFundsError if there is not enough.
    withdraw(amount) {
        if (amount > this.balance) {
            throw new InsufficientFundsError("There is not enough money in the account");
        }
        this.balance -= amount;
    }

    // Returns the account as a text, ready to show.
    toString() {
        return `${this.owner}: ${this.balance.toFixed(2)} €`;
    }
}

// Exercise 24.3: our own kind of error
class InsufficientFundsError extends Error {
    constructor(message) {
        super(message);
        this.name = "InsufficientFundsError";
    }
}

const account = new BankAccount("Ana", 100);
const accountOutput = document.querySelector("#account");
const message = document.querySelector("#message");

// Draws the account in the page.
function draw() {
    accountOutput.textContent = account.toString();
}

document.querySelector("#depositButton").addEventListener("click", function () {
    const amount = Number(document.querySelector("#amount").value);
    account.deposit(amount);
    message.textContent = "";
    draw();
});

document.querySelector("#withdrawButton").addEventListener("click", function () {
    const amount = Number(document.querySelector("#amount").value);
    try {
        account.withdraw(amount);
        message.textContent = "";
        message.className = "";
    } catch (error) {
        if (error instanceof InsufficientFundsError) {
            message.textContent = "Not enough money. Try with a smaller amount.";
        } else {
            message.textContent = "Something unexpected happened.";
        }
        message.className = "error";
    }
    draw();
});

draw();

Fitxer de partida: exercises/24-inheritance/bank.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

InsufficientFundsError extends Error és un error propi; withdraw el llança i en el catch es distingix amb instanceof d'un error inesperat. Cal definir la classe d'error abans d'usar-la.

Exercici 24.4

Exercise 24.4. An array with cars and motorbikes mixed: go through it calling describe() on each one. Why does each one use its own version?

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Vehicles</title>
</head>

<body>
    <h1>Vehicles</h1>
    <ul id="list"></ul>

    <script>
        class Vehicle {
            constructor(brand, model) {
                this.brand = brand;
                this.model = model;
            }

            describe() {
                return `${this.brand} ${this.model}`;
            }
        }

        class Car extends Vehicle {
            constructor(brand, model, doors) {
                super(brand, model);         // first: the constructor of Vehicle
                this.doors = doors;
            }

            describe() {
                return `${super.describe()}, ${this.doors} doors`;
            }
        }

        class Motorbike extends Vehicle {
            constructor(brand, model, engineSize) {
                super(brand, model);
                this.engineSize = engineSize;
            }

            describe() {
                return `${super.describe()}, ${this.engineSize} cc`;
            }
        }

        const vehicles = [
            new Car("Seat", "Ibiza", 5),
            new Motorbike("Yamaha", "MT-07", 689),
            new Car("Mini", "Cooper", 3),
            new Motorbike("Honda", "CBR", 600)
        ];

        for (const vehicle of vehicles) {
            const li = document.createElement("li");
            li.textContent = vehicle.describe();       // each one uses its own describe()
            document.querySelector("#list").appendChild(li);
        }
    </script>
</body>

</html>
Resposta: cada objecte usa la seua pròpia versió de describe() perquè JavaScript busca el mètode primer en la classe de l'objecte (Car o Motorbike) i només puja a Vehicle si no el troba. Això es diu polimorfisme: el mateix codi (vehicle.describe()) es comporta diferent segons el tipus real de l'objecte.
