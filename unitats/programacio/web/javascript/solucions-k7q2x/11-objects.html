Exercici 11.1

Exercise 11.1. Create an object for your favourite video game: title, studio, year, rating, and whether it is multiplayer. Draw it in the page as a card.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>My favourite game</title>
    <style>
        .card {
            display: inline-block;
            padding: 12px 20px;
            border: 2px solid #1f3c5a;
            border-radius: 10px;
        }
    </style>
</head>

<body>
    <h1>My favourite game</h1>
    <div id="card"></div>

    <script>
        const game = {
            title: "Minecraft",
            studio: "Mojang",
            year: 2011,
            rating: 9.5,
            multiplayer: true
        };

        document.querySelector("#card").innerHTML = `
            <div class="card">
                <h2>${game.title}</h2>
                <p>Studio: ${game.studio}</p>
                <p>Year: ${game.year}</p>
                <p>Rating: ${game.rating}</p>
                <p>Multiplayer: ${game.multiplayer ? "yes" : "no"}</p>
            </div>`;
    </script>
</body>

</html>

Exercici 11.2

Exercise 11.2. Create an array with 5 products (name, price, stock) and draw one card for each one.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products</title>
    <style>
        #cards {
            display: flex;
            flex-wrap: wrap;
            gap: 12px;
        }

        .card {
            width: 160px;
            padding: 12px;
            border: 2px solid #cccccc;
            border-radius: 8px;
        }

        .cheapest {
            border-color: gold;
        }
    </style>
</head>

<body>
    <h1>Products</h1>
    <div id="cards"></div>

    <script>
        const products = [
            { name: "Keyboard", price: 25, stock: 10 },
            { name: "Mouse", price: 15, stock: 0 },
            { name: "Monitor", price: 120, stock: 3 },
            { name: "Cable", price: 5, stock: 40 },
            { name: "Webcam", price: 45, stock: 0 }

        function draw(list) {
            let html = "";
            for (const p of list) {
                html += `<div class="card"><h3>${p.name}</h3><p>${p.price} euros</p><p>Stock: ${p.stock}</p></div>`;
            }
            document.querySelector("#cards").innerHTML = html;
        }

        draw(products);
    </script>
</body>

</html>

Exercici 11.4

Exercise 11.4. Write a function cheapest(products) that returns the object of the cheapest product. Highlight that card with a coloured border.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products</title>
    <style>
        #cards {
            display: flex;
            flex-wrap: wrap;
            gap: 12px;
        }

        .card {
            width: 160px;
            padding: 12px;
            border: 2px solid #cccccc;
            border-radius: 8px;
        }

        .cheapest {
            border-color: gold;
        }
    </style>
</head>

<body>
    <h1>Products</h1>
    <div id="cards"></div>

    <script>
        const products = [
            { name: "Keyboard", price: 25, stock: 10 },
            { name: "Mouse", price: 15, stock: 0 },
            { name: "Monitor", price: 120, stock: 3 },
            { name: "Cable", price: 5, stock: 40 },
            { name: "Webcam", price: 45, stock: 0 }
        ];

        function cheapest(list) {
            let best = list[0];
            for (const p of list) {
                if (p.price < best.price) {
                    best = p;
                }
            }
            return best;
        }

        function draw(list) {
            const cheap = cheapest(list);
            let html = "";
            for (const p of list) {
                const extra = p === cheap ? " cheapest" : "";
                html += `<div class="card${extra}"><h3>${p.name}</h3><p>${p.price} euros</p><p>Stock: ${p.stock}</p></div>`;
            }
            document.querySelector("#cards").innerHTML = html;
        }

        draw(products);
    </script>
</body>

</html>

Exercici 11.5

Exercise 11.5. Write a function outOfStock(products) that returns an array with the names of the products with stock 0. Show them in a red list.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products</title>
    <style>
        #cards {
            display: flex;
            flex-wrap: wrap;
            gap: 12px;
        }

        .card {
            width: 160px;
            padding: 12px;
            border: 2px solid #cccccc;
            border-radius: 8px;
        }

        .cheapest {
            border-color: gold;
        }
    </style>
</head>

<body>
    <h1>Products</h1>
    <div id="cards"></div>
    <h2>Out of stock</h2>
    <ul id="out"></ul>

    <script>
        const products = [
            { name: "Keyboard", price: 25, stock: 10 },
            { name: "Mouse", price: 15, stock: 0 },
            { name: "Monitor", price: 120, stock: 3 },
            { name: "Cable", price: 5, stock: 40 },
            { name: "Webcam", price: 45, stock: 0 }
        ];

        function draw(list) {
            let html = "";
            for (const p of list) {
                html += `<div class="card"><h3>${p.name}</h3><p>${p.price} euros</p><p>Stock: ${p.stock}</p></div>`;
            }
            document.querySelector("#cards").innerHTML = html;
        }

        function outOfStock(list) {
            const names = [];
            for (const p of list) {
                if (p.stock === 0) {
                    names.push(p.name);
                }
            }
            return names;
        }

        draw(products);

        let html = "";
        for (const name of outOfStock(products)) {
            html += `<li style="color: red">${name}</li>`;
        }
        document.querySelector("#out").innerHTML = html;
    </script>
</body>

</html>

Exercici 11.6

Exercise 11.6. Two buttons: "Sort by price" and "Sort by name". Each one sorts the array and redraws the cards.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products</title>
    <style>
        #cards {
            display: flex;
            flex-wrap: wrap;
            gap: 12px;
        }

        .card {
            width: 160px;
            padding: 12px;
            border: 2px solid #cccccc;
            border-radius: 8px;
        }

        .cheapest {
            border-color: gold;
        }
    </style>
</head>

<body>
    <h1>Products</h1>
    <button id="sortByPrice">Sort by price</button>
    <button id="sortByName">Sort by name</button>
    <div id="cards"></div>

    <script>
        const products = [
            { name: "Keyboard", price: 25, stock: 10 },
            { name: "Mouse", price: 15, stock: 0 },
            { name: "Monitor", price: 120, stock: 3 },
            { name: "Cable", price: 5, stock: 40 },
            { name: "Webcam", price: 45, stock: 0 }
        ];

        function draw(list) {
            let html = "";
            for (const p of list) {
                html += `<div class="card"><h3>${p.name}</h3><p>${p.price} euros</p><p>Stock: ${p.stock}</p></div>`;
            }
            document.querySelector("#cards").innerHTML = html;
        }

        document.querySelector("#sortByPrice").addEventListener("click", function () {
            products.sort((a, b) => a.price - b.price);
            draw(products);
        });

        document.querySelector("#sortByName").addEventListener("click", function () {
            products.sort((a, b) => a.name.localeCompare(b.name));
            draw(products);
        });

        draw(products);
    </script>
</body>

</html>
sort modifica l'array original; per ordenar números cal una funció de comparació (a.price - b.price) i per al text, localeCompare.

Exercici 11.7

Exercise 11.7. Class list. An array of students with name, age and mark. Draw a table. Marks under 5 appear in red. Under the table: average, best student and how many passed.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Class list</title>
</head>

<body>
    <h1>Class list</h1>
    <table id="table"></table>
    <p id="summary"></p>

    <script>
        const students = [
            { name: "Pere", age: 16, mark: 7 },
            { name: "Joan", age: 17, mark: 4 },
            { name: "Marta", age: 16, mark: 9 },
            { name: "Nuria", age: 18, mark: 3.5 },
            { name: "Laia", age: 16, mark: 6 }
        ];

        let html = "<tr><th>Name</th><th>Age</th><th>Mark</th></tr>";
        let sum = 0;
        let passed = 0;
        let best = students[0];

        for (const s of students) {
            const colour = s.mark < 5 ? "red" : "black";
            html += `<tr><td>${s.name}</td><td>${s.age}</td><td style="color: ${colour}">${s.mark}</td></tr>`;
            sum += s.mark;
            if (s.mark >= 5) {
                passed++;
            }
            if (s.mark > best.mark) {
                best = s;
            }
        }

        document.querySelector("#table").innerHTML = html;
        document.querySelector("#summary").textContent =
            `Average: ${(sum / students.length).toFixed(2)} | Best student: ${best.name} | Passed: ${passed}`;
    </script>
</body>

</html>
