Exercici 13.1

Exercise 13.1. Given an array of 8 strings, show next to each one whether it is a valid postcode.

Solució

script.js 
function show(text) {
    document.body.insertAdjacentHTML("beforeend", "<p>" + text + "</p>");
}

const postcodes = ["46001", "4600", "46 001", "46001A", "08080", "12345", "ABCDE", "460011"];
const pattern = /^[0-9]{5}$/;

for (const text of postcodes) {
    show(`${text} -> ${pattern.test(text) ? "valid" : "NOT valid"}`);
}
^ i $ obliguen que tot el text siga exactament 5 dígits (sense ells, "460011" també coincidiria).

Exercici 13.2

Exercise 13.2. The same with a simple email pattern. Include tricky cases: "ana@", "@mail.com", "ana@mail".

Solució

script.js 
function show(text) {
    document.body.insertAdjacentHTML("beforeend", "<p>" + text + "</p>");
}

const emails = ["ana@mail.com", "ana@", "@mail.com", "ana@mail", "ana.garcia@school.org", "ana @mail.com"];
const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

for (const text of emails) {
    show(`${text} -> ${pattern.test(text) ? "valid" : "NOT valid"}`);
}
El patró vol: text sense espais ni @, una @, més text, un punt i més text. Per això "ana@", "@mail.com" i "ana@mail" fallen.

Exercici 13.3

Exercise 13.3. Redo the password check of exercise 5.3 with three regular expressions: at least 8 characters, a number, a capital letter.

Solució

script.js 
const input = document.querySelector("#password");
const result = document.querySelector("#result");

function addLine(ok, text) {
    const line = document.createElement("p");
    line.className = ok ? "ok" : "ko";
    line.textContent = (ok ? "\u2714 " : "\u2718 ") + text;
    line.style.color = ok ? "green" : "red";
    result.appendChild(line);
}

document.querySelector("#checkButton").addEventListener("click", function () {
    const password = input.value;
    result.textContent = "";

    addLine(/.{8,}/.test(password), "At least 8 characters");
    addLine(/[0-9]/.test(password), "Has a number");
    addLine(/[A-Z]/.test(password), "Has a capital letter");
});

Exercici 13.4

Exercise 13.4. Validate a DNI (8 digits and a letter) and show the letter alone, taken out with match.

Solució

script.js 
function show(text) {
    document.body.insertAdjacentHTML("beforeend", "<p>" + text + "</p>");
}

const dnis = ["12345678Z", "1234567Z", "12345678", "12345678z", "ABCDEFGH1"];
const pattern = /^[0-9]{8}[A-Za-z]$/;

for (const dni of dnis) {
    if (pattern.test(dni)) {
        const letter = dni.match(/[A-Za-z]$/)[0];
        show(`${dni} -> valid, the letter is ${letter}`);
    } else {
        show(`${dni} -> NOT valid`);
    }
}

Exercici 13.5

Exercise 13.5. A phone written as "961-23 45 67": remove every space and dash with replace.

Solució

script.js 
function show(text) {
    document.body.insertAdjacentHTML("beforeend", "<p>" + text + "</p>");
}

const phone = "961-23 45 67";
const clean = phone.replace(/[\s-]/g, "");   // g: all of them, not only the first

show(`${phone} -> ${clean}`);
La bandera g fa que es substituïsquen tots els espais i guionets, no només el primer.

Exercici 13.6

Exercise 13.6. Take all the numbers out of a sentence with match and the flag g, and add them up.

Solució

script.js 
function show(text) {
    document.body.insertAdjacentHTML("beforeend", "<p>" + text + "</p>");
}

const sentence = "I bought 3 apples, 12 oranges and 5 bananas for 20 euros.";
const numbers = sentence.match(/[0-9]+/g);        // ["3", "12", "5", "20"]

let total = 0;
for (const n of numbers) {
    total += Number(n);
}

show(`Numbers found: ${numbers.join(", ")}`);
show(`Sum: ${total}`);
