Exercici 23.1

Exercise 23.1. Change your BankAccount so that the balance is #balance, with a get balance(). Try to read account.#balance from outside and copy the error.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Private balance</title>
</head>

<body>
    <h1>Private balance</h1>
    <p id="output"></p>

    <script>
        class BankAccount {
            #balance = 0;

            constructor(owner) {
                this.owner = owner;
            }

            deposit(amount) {
                this.#balance += amount;
            }

            withdraw(amount) {
                if (amount > this.#balance) {
                    throw new Error("There is not enough money in the account");
                }
                this.#balance -= amount;
            }

            get balance() {
                return this.#balance;
            }
        }

        const account = new BankAccount("Ana");
        account.deposit(50);
        document.querySelector("#output").textContent = `${account.owner}: ${account.balance} euros`;

        // account.#balance;   // SyntaxError: Private field '#balance' must be declared in an enclosing class
    </script>
</body>

</html>
Resposta: si es posa account.#balance fora de la classe, el navegador ni executa el fitxer: Uncaught SyntaxError: Private field '#balance' must be declared in an enclosing class. Des de fora només es pot llegir amb el get balance (account.balance) i canviar-lo amb els mètodes deposit i withdraw.

Exercici 23.2

Exercise 23.2. Add to Product a set price that throws an error with negative prices. Check it with try/catch.

Solució

script.js 
class Product {
    #price;

    constructor(name, price) {
        this.name = name;
        this.price = price;               // goes through the set
    }

    get price() {
        return this.#price;
    }

    set price(value) {
        if (value < 0) {
            throw new Error("The price cannot be negative");
        }
        this.#price = value;
    }
}

const mouse = new Product("Mouse", 15);
console.log("Price: " + mouse.price);

try {
    mouse.price = -5;                     // the set throws the error
} catch (error) {
    console.log("Error: " + error.message);
}

console.log("Price after the error: " + mouse.price);   // still 15

Exercici 23.3

Exercise 23.3. Save an array of Product in localStorage and, when the page loads, rebuild the objects with Product.fromJSON. Check that their methods work after F5.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Saved products</title>
</head>

<body>
    <h1>Saved products</h1>
    <input type="text" id="nameInput" placeholder="Name">
    <input type="number" id="priceInput" placeholder="Price">
    <input type="number" id="stockInput" placeholder="Stock">
    <button id="addButton">Add</button>
    <ul id="list"></ul>

    <script>
        class Product {
            #price;
            static count = 0;                     // static property: belongs to the class

            constructor(name, price, stock = 0) {
                this.name = name;
                this.price = price;               // goes through the set
                this.stock = stock;
                Product.count++;
            }

            get price() {
                return this.#price;
            }

            set price(value) {
                if (value < 0) {
                    throw new Error("The price cannot be negative");
                }
                this.#price = value;
            }

            stockValue() {
                return this.price * this.stock;
            }

            // from plain data (JSON) to a Product again
            static fromJSON(data) {
                return new Product(data.name, data.price, data.stock);
            }

            // what is saved when we do JSON.stringify
            toJSON() {
                return { name: this.name, price: this.#price, stock: this.stock };
            }
        }

        // Load: plain data -> Product objects again (so the methods work after F5)
        function load() {
            try {
                const data = JSON.parse(localStorage.getItem("products")) || [];
                return data.map(d => Product.fromJSON(d));
            } catch (error) {
                return [];
            }
        }

        let products = load();

        function save() {
            localStorage.setItem("products", JSON.stringify(products));   // uses toJSON of each one
        }

        function draw() {
            const list = document.querySelector("#list");
            list.textContent = "";

            for (const p of products) {
                const li = document.createElement("li");
                li.textContent = `${p.name}: ${p.price} euros x ${p.stock} = ${p.stockValue()} euros`;
                list.appendChild(li);
            }
        }

        document.querySelector("#addButton").addEventListener("click", function () {
            try {
                const name = document.querySelector("#nameInput").value.trim();
                const price = Number(document.querySelector("#priceInput").value);
                const stock = Number(document.querySelector("#stockInput").value);
                products.push(new Product(name, price, stock));
                save();
                draw();
            } catch (error) {
                alert(error.message);
            }
        });

        draw();
    </script>
</body>

</html>
Sense fromJSON, després de JSON.parse els elements serien objectes simples i p.stockValue() donaria error. Amb Product.fromJSON tornen a ser objectes Product amb tots els seus mètodes. En la previsualització el localStorage està simulat en memòria.

Exercici 23.4

Exercise 23.4. A static property Product.count that goes up every time a product is created.

Solució

script.js 
class Product {
    static count = 0;                     // belongs to the class, not to each product

    constructor(name, price) {
        this.name = name;
        this.price = price;
        Product.count++;                  // goes up every time a product is created
    }
}

console.log("Products: " + Product.count);       // 0
new Product("Mouse", 15);
new Product("Keyboard", 25);
new Product("Cable", 5);
console.log("Products: " + Product.count);       // 3
