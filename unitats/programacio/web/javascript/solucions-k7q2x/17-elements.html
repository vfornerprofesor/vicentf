Exercici 17.1

Exercise 17.1. Rewrite exercise 16.4 (the add-item list) using createElement instead of innerHTML.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Add items</title>
</head>

<body>
    <h1>My list</h1>
    <input type="text" id="nameInput" placeholder="Something">
    <button id="addButton">Add</button>
    <ul id="list"></ul>

    <script>
        const input = document.querySelector("#nameInput");
        const list = document.querySelector("#list");
        const items = [];

        function draw() {
            list.textContent = "";
            for (const text of items) {
                const li = document.createElement("li");
                li.textContent = text;
                list.appendChild(li);
            }
        }

        function addItem() {
            const text = input.value.trim();
            if (text === "") {
                return;
            }
            items.push(text);
            input.value = "";
            input.focus();
            draw();
        }

        document.querySelector("#addButton").addEventListener("click", addItem);
        input.addEventListener("keydown", function (event) {
            if (event.key === "Enter") {
                addItem();
            }
        });
    </script>
</body>

</html>

Exercici 17.2

Exercise 17.2. Add a "Delete" button to every row, that removes that item from the array and redraws.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Delete items</title>
</head>

<body>
    <h1>My list</h1>
    <input type="text" id="nameInput" placeholder="Something">
    <button id="addButton">Add</button>
    <ul id="list"></ul>
    <p id="counter"></p>

    <script>
        const input = document.querySelector("#nameInput");
        const list = document.querySelector("#list");
        const counter = document.querySelector("#counter");
        const items = [];

        function draw() {
            list.textContent = "";                         // empty the list

            items.forEach(function (text, index) {
                const li = document.createElement("li");
                li.textContent = text;                     // textContent: safe with any text

                const remove = document.createElement("button");
                remove.textContent = "Delete";
                remove.addEventListener("click", function () {
                    items.splice(index, 1);                // remove it from the array
                    draw();                                // and redraw
                });

                li.appendChild(remove);
                list.appendChild(li);
            });

            counter.textContent = items.length + (items.length === 1 ? " item" : " items");
        }

        function addItem() {
            const text = input.value.trim();
            if (text === "") {
                return;
            }
            items.push(text);
            input.value = "";
            draw();
        }

        document.querySelector("#addButton").addEventListener("click", addItem);
        input.addEventListener("keydown", function (event) {
            if (event.key === "Enter") {
                addItem();
            }
        });

        draw();
    </script>
</body>

</html>
El index de cada fila es guarda en la funció del botó i splice(index, 1) lleva eixe element de l'array; després es redibuixa tot.

Exercici 17.4

Exercise 17.4. A "Gallery" page: an array of 6 image URLs, and a loop that creates 6 <img> elements with createElement. Clicking on one makes it bigger.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Gallery</title>
    <style>
        img {
            width: 200px;
            margin: 6px;
            cursor: zoom-in;
            transition: width 0.3s ease;
        }

        img.big {
            width: 420px;
            cursor: zoom-out;
        }
    </style>
</head>

<body>
    <h1>Gallery</h1>
    <div id="gallery"></div>

    <script>
        const urls = [
            "https://picsum.photos/seed/a/200/150",
            "https://picsum.photos/seed/b/200/150",
            "https://picsum.photos/seed/c/200/150",
            "https://picsum.photos/seed/d/200/150",
            "https://picsum.photos/seed/e/200/150",
            "https://picsum.photos/seed/f/200/150"
        ];

        const gallery = document.querySelector("#gallery");

        for (const url of urls) {
            const img = document.createElement("img");
            img.src = url;
            img.alt = "A random photo";
            img.addEventListener("click", function () {
                img.classList.toggle("big");        // clicking again makes it small
            });
            gallery.appendChild(img);
        }
    </script>
</body>

</html>
Les imatges vénen de picsum.photos, per tant cal connexió a internet per a veure-les.

Exercici 17.5

Exercise 17.5. A table of students created with createElement: one <tr> per student, one <td> per property.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Students table</title>
    <style>
        td {
            padding: 4px 12px;
            border: 1px solid #999999;
        }
    </style>
</head>

<body>
    <h1>Students</h1>
    <table id="table"></table>

    <script>
        const students = [
            { name: "Pere", age: 16, mark: 7 },
            { name: "Joan", age: 17, mark: 4 },
            { name: "Marta", age: 16, mark: 9 }
        ];

        const table = document.querySelector("#table");

        for (const student of students) {
            const tr = document.createElement("tr");

            for (const key in student) {              // one td per property
                const td = document.createElement("td");
                td.textContent = student[key];
                tr.appendChild(td);
            }

            table.appendChild(tr);
        }
    </script>
</body>

</html>

Exercici 17.6

Exercise 17.6. A "message box": a function showMessage(text, type) that creates a <div> with the class ok or error, puts it at the top of the page, and removes it by itself after 3 seconds. (setTimeout(function () { ... }, 3000))

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Message box</title>
    <style>
        .message {
            padding: 10px;
            margin-bottom: 8px;
            border-radius: 6px;
        }

        .ok {
            background-color: #d4f5dd;
            color: #146c2e;
        }

        .error {
            background-color: #fde2e2;
            color: #a11d1d;
        }
    </style>
</head>

<body>
    <button id="ok">Ok message</button>
    <button id="error">Error message</button>

    <script>
        function showMessage(text, type) {
            const box = document.createElement("div");
            box.textContent = text;
            box.classList.add("message", type);

            document.body.prepend(box);

            setTimeout(function () {
                box.remove();
            }, 3000);
        }

        document.querySelector("#ok").addEventListener("click", function () {
            showMessage("Task added", "ok");
        });

        document.querySelector("#error").addEventListener("click", function () {
            showMessage("Something went wrong", "error");
        });
    </script>
</body>

</html>
La prova clica i espera 3,1 segons: el missatge s'ha llevat sol.

Mini-projecte 2

Mini-project 2: A list that works

One page with a list of anything you like (films to watch, songs, shopping, homework). It must:

  • keep the data in an array;
  • add an item with a button and with Enter;
  • refuse an empty item;
  • draw the list with createElement, never with innerHTML for the user's text;
  • have a Delete button on every row that changes the array and redraws;
  • show a counter of items;
  • have a "Clear all" button that asks for confirmation (confirm("Are you sure?")).

No localStorage yet: that comes in lesson 19, and then you will add it to this same page.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>A list that works</title>
</head>

<body>
    <h1>Films to watch</h1>
    <input type="text" id="filmInput" placeholder="A film">
    <button id="addButton">Add</button>
    <button id="clearButton">Clear all</button>
    <ul id="list"></ul>
    <p id="counter"></p>

    <script>
        const films = [];
        const input = document.querySelector("#filmInput");
        const list = document.querySelector("#list");
        const counter = document.querySelector("#counter");

        function draw() {
            list.textContent = "";

            films.forEach(function (film, index) {
                const li = document.createElement("li");
                li.textContent = film;                       // never innerHTML with the user's text

                const remove = document.createElement("button");
                remove.textContent = "Delete";
                remove.addEventListener("click", function () {
                    films.splice(index, 1);
                    draw();
                });

                li.appendChild(remove);
                list.appendChild(li);
            });

            counter.textContent = films.length + (films.length === 1 ? " film" : " films");
        }

        function addFilm() {
            const text = input.value.trim();
            if (text === "") {
                return;                                      // refuse an empty item
            }
            films.push(text);
            input.value = "";
            input.focus();
            draw();
        }

        document.querySelector("#addButton").addEventListener("click", addFilm);

        input.addEventListener("keydown", function (event) {
            if (event.key === "Enter") {
                addFilm();
            }
        });

        document.querySelector("#clearButton").addEventListener("click", function () {
            if (confirm("Are you sure?")) {
                films.length = 0;
                draw();
            }
        });

        draw();
    </script>
</body>

</html>
Compleix tots els requisits: array, afegir amb botó i amb Enter, no accepta buit, createElement, botó Delete per fila que canvia l'array i redibuixa, comptador i "Clear all" amb confirm. (En la previsualització el confirm sempre respon que sí.)
