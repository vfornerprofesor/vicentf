Exercici 22.1

Exercise 22.1. Write the class Student above, create three students and show them in the page with toString().

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Students</title>
</head>

<body>
    <h1>Students</h1>
    <ul id="list"></ul>

    <script>
        class Student {
            constructor(name, marks) {
                this.name = name;
                this.marks = marks;
            }

            average() {
                const total = this.marks.reduce((sum, m) => sum + m, 0);
                return total / this.marks.length;
            }

            toString() {
                return `${this.name}: ${this.average().toFixed(2)}`;
            }
        }

        const students = [
            new Student("Ana", [7, 8, 9]),
            new Student("Luis", [4, 6, 5]),
            new Student("Marta", [10, 9, 9])
        ];

        const list = document.querySelector("#list");

        for (const student of students) {
            const li = document.createElement("li");
            li.textContent = student.toString();
            list.appendChild(li);
        }
    </script>
</body>

</html>

Exercici 22.2

Exercise 22.2. Redo exercise 11.2 with a class Product (name, price, stock) and a method stockValue(). The array is now an array of Product objects, drawn as cards.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products</title>
    <style>
        .card {
            display: inline-block;
            width: 160px;
            margin: 6px;
            padding: 10px;
            border: 2px solid #cccccc;
            border-radius: 8px;
        }
    </style>
</head>

<body>
    <h1>Products</h1>
    <div id="cards"></div>

    <script>
        class Product {
            constructor(name, price, stock) {
                this.name = name;
                this.price = price;
                this.stock = stock;
            }

            // value of the stock of this product
            stockValue() {
                return this.price * this.stock;
            }
        }

        const products = [
            new Product("Keyboard", 25, 10),
            new Product("Mouse", 15, 0),
            new Product("Monitor", 120, 3),
            new Product("Cable", 5, 40),
            new Product("Webcam", 45, 0)
        ];

        let html = "";
        for (const p of products) {
            html += `<div class="card"><h3>${p.name}</h3><p>${p.price} euros</p><p>Stock value: ${p.stockValue()} euros</p></div>`;
        }
        document.querySelector("#cards").innerHTML = html;
    </script>
</body>

</html>

Exercici 22.3

Exercise 22.3. A class BankAccount with owner and balance, and the methods deposit(amount) and withdraw(amount). If there is not enough money, withdraw throws an error. A small page with two buttons uses it.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Bank account</title>
</head>

<body>
    <h1>Bank account</h1>
    <input type="number" id="amount" placeholder="Amount in euros">
    <button id="deposit">Deposit</button>
    <button id="withdraw">Withdraw</button>
    <p id="account"></p>
    <p id="message"></p>

    <script>
        class BankAccount {
            constructor(owner, balance) {
                this.owner = owner;
                this.balance = balance;
            }

            deposit(amount) {
                this.balance += amount;
            }

            // throws an error if there is not enough money
            withdraw(amount) {
                if (amount > this.balance) {
                    throw new Error("There is not enough money in the account");
                }
                this.balance -= amount;
            }
        }

        const account = new BankAccount("Ana", 100);

        function draw() {
            document.querySelector("#account").textContent = `${account.owner}: ${account.balance.toFixed(2)} euros`;
        }

        document.querySelector("#deposit").addEventListener("click", function () {
            account.deposit(Number(document.querySelector("#amount").value));
            document.querySelector("#message").textContent = "";
            draw();
        });

        document.querySelector("#withdraw").addEventListener("click", function () {
            const message = document.querySelector("#message");
            try {
                account.withdraw(Number(document.querySelector("#amount").value));
                message.textContent = "";
            } catch (error) {
                message.textContent = error.message;
                message.style.color = "red";
            }
            draw();
        });

        draw();
    </script>
</body>

</html>

Exercici 22.4

Exercise 22.4. Add toString() to Product and BankAccount and use it to show them.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Products and account</title>
</head>

<body>
    <h1>Products and account</h1>
    <p id="product"></p>
    <p id="account"></p>

    <script>
        class Product {
            constructor(name, price, stock) {
                this.name = name;
                this.price = price;
                this.stock = stock;
            }

            // value of the stock of this product
            stockValue() {
                return this.price * this.stock;
            }

            toString() {
                return `${this.name}: ${this.price} euros, ${this.stock} in stock`;
            }
        }

        class BankAccount {
            constructor(owner, balance) {
                this.owner = owner;
                this.balance = balance;
            }

            deposit(amount) {
                this.balance += amount;
            }

            // throws an error if there is not enough money
            withdraw(amount) {
                if (amount > this.balance) {
                    throw new Error("There is not enough money in the account");
                }
                this.balance -= amount;
            }

            toString() {
                return `${this.owner}: ${this.balance.toFixed(2)} euros`;
            }
        }

        const mouse = new Product("Mouse", 15, 4);
        const account = new BankAccount("Ana", 100);

        document.querySelector("#product").textContent = mouse.toString();
        document.querySelector("#account").textContent = account.toString();
    </script>
</body>

</html>
Quan es posa un objecte on es necessita un text (per exemple, en una plantilla `${mouse}`), JavaScript crida automàticament a toString().
