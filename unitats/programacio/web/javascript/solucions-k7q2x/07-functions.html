Exercici 7.1

Exercise 7.1. Write a function square(n) that returns n * n. Add an input and a button so the user can test it with any number.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Square</title>
</head>

<body>
    <input type="number" id="n" placeholder="A number">
    <button id="go">Square</button>
    <p id="out"></p>

    <script>
        function square(n) {
            return n * n;
        }

        document.querySelector("#go").addEventListener("click", function () {
            const n = Number(document.querySelector("#n").value);
            document.querySelector("#out").textContent = `${n} squared is ${square(n)}`;
        });
    </script>
</body>

</html>

Exercici 7.2

Exercise 7.2. Write a function isEven(n) that returns true or false. Show the answer in the page as "Even" in green or "Odd" in orange.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Even or odd</title>
</head>

<body>
    <input type="number" id="n" placeholder="A number">
    <button id="go">Check</button>
    <p id="out"></p>

    <script>
        function isEven(n) {
            return n % 2 === 0;
        }

        document.querySelector("#go").addEventListener("click", function () {
            const n = Number(document.querySelector("#n").value);
            const out = document.querySelector("#out");

            if (isEven(n)) {
                out.textContent = "Even";
                out.style.color = "green";
            } else {
                out.textContent = "Odd";
                out.style.color = "orange";
            }
        });
    </script>
</body>

</html>

Exercici 7.3

Exercise 7.3. Write a function biggest(a, b, c) that returns the biggest of three numbers. Three inputs, one button.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Biggest</title>
</head>

<body>
    <input type="number" id="a" placeholder="A">
    <input type="number" id="b" placeholder="B">
    <input type="number" id="c" placeholder="C">
    <button id="go">Biggest</button>
    <p id="out"></p>

    <script>
        function biggest(a, b, c) {
            let max = a;
            if (b > max) {
                max = b;
            }
            if (c > max) {
                max = c;
            }
            return max;
        }

        document.querySelector("#go").addEventListener("click", function () {
            const a = Number(document.querySelector("#a").value);
            const b = Number(document.querySelector("#b").value);
            const c = Number(document.querySelector("#c").value);
            document.querySelector("#out").textContent = `The biggest is ${biggest(a, b, c)}`;
        });
    </script>
</body>

</html>

Exercici 7.4

Exercise 7.4. Calculator. Build the calculator below. One function per operation, all of them returning a value, plus one function that reads the inputs and shows the result. No repeated code.

Fitxer de partida: ex4.html

Solució

script.js 
function add(a, b) {
    return a + b;
}

function subtract(a, b) {
    return a - b;
}

function multiply(a, b) {
    return a * b;
}

function divide(a, b) {
    if (b === 0) {
        return "You cannot divide by zero";
    }
    return a / b;
}

// ONE function that reads the inputs and shows the result
function calculate() {
    const a = Number(document.querySelector("#firstNumber").value);
    const b = Number(document.querySelector("#secondNumber").value);
    const operation = document.querySelector("#operation").value;
    let result;

    if (operation === "+") {
        result = add(a, b);
    } else if (operation === "-") {
        result = subtract(a, b);
    } else if (operation === "*") {
        result = multiply(a, b);
    } else {
        result = divide(a, b);
    }

    document.querySelector("#result").textContent = result;
}

document.querySelector("#calculateButton").addEventListener("click", calculate);

Fitxer de partida: exercises/07-functions/ex4.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Cada operació és una funció que retorna un valor; calculate només llig, crida a la funció que toca i mostra. No es repeteix cap càlcul.

Exercici 7.5

Exercise 7.5. Write celsiusToFahrenheit(c) and fahrenheitToCelsius(f). One page with two converters, side by side. Check that one undoes the other.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Temperature converters</title>
</head>

<body>
    <p>
        <input type="number" id="c" placeholder="Celsius">
        <button id="toF">To Fahrenheit</button>
        <span id="outF"></span>
    </p>
    <p>
        <input type="number" id="f" placeholder="Fahrenheit">
        <button id="toC">To Celsius</button>
        <span id="outC"></span>
    </p>

    <script>
        function celsiusToFahrenheit(c) {
            return c * 9 / 5 + 32;
        }

        function fahrenheitToCelsius(f) {
            return (f - 32) * 5 / 9;
        }

        document.querySelector("#toF").addEventListener("click", function () {
            const c = Number(document.querySelector("#c").value);
            document.querySelector("#outF").textContent = celsiusToFahrenheit(c) + " F";
        });

        document.querySelector("#toC").addEventListener("click", function () {
            const f = Number(document.querySelector("#f").value);
            document.querySelector("#outC").textContent = fahrenheitToCelsius(f) + " C";
        });

        // one undoes the other: 25 -> 77 -> 25
        console.log(fahrenheitToCelsius(celsiusToFahrenheit(25)));
    </script>
</body>

</html>
La línia final comprova que una funció desfà l'altra: fahrenheitToCelsius(celsiusToFahrenheit(25)) torna 25.

Exercici 7.6

Exercise 7.6. Write a function finalPrice(price, discount) that returns the price with the discount applied. If no discount is given, it should be 0. (Look up "default parameter".)

Solució

script.js 
function finalPrice(price, discount = 0) {
    return price - (price * discount / 100);
}

console.log(finalPrice(100, 20));   // 80
console.log(finalPrice(100));       // 100  (no discount: the default value 0 is used)
El valor per defecte (discount = 0) s'usa quan la funció es crida sense eixe paràmetre.

Exercici 7.7

Exercise 7.7. Write a function countVowels(text) that returns how many vowels a text has. The user types a sentence in an input and the page shows the count as they click the button.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Count vowels</title>
</head>

<body>
    <input type="text" id="text" placeholder="Write a sentence">
    <button id="go">Count</button>
    <p id="out"></p>

    <script>
        function countVowels(text) {
            const vowels = "aeiou";
            let count = 0;

            for (const letter of text.toLowerCase()) {
                if (vowels.includes(letter)) {
                    count++;
                }
            }
            return count;
        }

        document.querySelector("#go").addEventListener("click", function () {
            const text = document.querySelector("#text").value;
            document.querySelector("#out").textContent = `The text has ${countVowels(text)} vowels.`;
        });
    </script>
</body>

</html>
