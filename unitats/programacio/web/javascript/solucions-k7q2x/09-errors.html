Exercici 9.1

Exercise 9.1. In your calculator of exercise 5.5, the division throws an error when the second number is 0. The button catches it and shows the message in red.

Solució

script.js 
const first = document.querySelector("#firstNumber");
const second = document.querySelector("#secondNumber");
const operation = document.querySelector("#operation");
const result = document.querySelector("#result");

function divide(a, b) {
    if (b === 0) {
        throw new Error("You cannot divide by zero");
    }
    return a / b;
}

document.querySelector("#calculateButton").addEventListener("click", function () {
    const a = Number(first.value);
    const b = Number(second.value);
    result.style.color = "";

    try {
        if (operation.value === "+") {
            result.textContent = a + b;
        } else if (operation.value === "-") {
            result.textContent = a - b;
        } else if (operation.value === "*") {
            result.textContent = a * b;
        } else {
            result.textContent = divide(a, b);   // can throw an error
        }
    } catch (error) {
        result.textContent = error.message;
        result.style.color = "red";
    }
});

Fitxer de partida: exercises/05-conditions/ex5.html (el codi de dalt va en el script.js o dins del <script> d'este fitxer).

Ara divide llança l'error amb throw en lloc de tornar un text, i el botó el captura amb try/catch.

Exercici 9.2

Exercise 9.2. Write readNumber(input): it returns the number in the field, or throws an error with name "ValidationError" and a different message when the field is empty, when it is not a number, or when it is outside a range you choose. Use it in exercises 7.1, 7.2 and 7.3.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Square with validation</title>
</head>

<body>
    <input type="text" id="n" placeholder="A number (0-1000)">
    <button id="go">Square</button>
    <p id="out"></p>

    <script>
        /**
         * Reads the number of a field.
         * @param {HTMLInputElement} input - The field.
         * @returns {number} The number typed by the user.
         * @throws {Error} ValidationError if it is empty, not a number or out of 0-1000.
         */
        function readNumber(input) {
            const text = input.value.trim();
            let error;

            if (text === "") {
                error = new Error("The field is empty.");
            } else if (isNaN(Number(text))) {
                error = new Error("It is not a number.");
            } else if (Number(text) < 0 || Number(text) > 1000) {
                error = new Error("The number must be between 0 and 1000.");
            }

            if (error) {
                error.name = "ValidationError";
                throw error;
            }
            return Number(text);
        }

        function square(n) {
            return n * n;
        }

        document.querySelector("#go").addEventListener("click", function () {
            const out = document.querySelector("#out");
            out.style.color = "";
            try {
                const n = readNumber(document.querySelector("#n"));
                out.textContent = `${n} squared is ${square(n)}`;
            } catch (error) {
                out.textContent = error.message;
                out.style.color = "red";
            }
        });
    </script>
</body>

</html>
readNumber llança un error amb name = "ValidationError" i un missatge diferent per a cada cas (buit, no és número, fora de rang). Es fa servir igual en 7.1, 7.2 i 7.3: només canvia la funció de càlcul.

Exercici 9.3

Exercise 9.3. Add finally to exercise 9.2 so that the field is always emptied.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Square with finally</title>
</head>

<body>
    <input type="text" id="n" placeholder="A number (0-1000)">
    <button id="go">Square</button>
    <p id="out"></p>

    <script>
        /**
         * Reads the number of a field.
         * @param {HTMLInputElement} input - The field.
         * @returns {number} The number typed by the user.
         * @throws {Error} ValidationError if it is empty, not a number or out of 0-1000.
         */
        function readNumber(input) {
            const text = input.value.trim();
            let error;

            if (text === "") {
                error = new Error("The field is empty.");
            } else if (isNaN(Number(text))) {
                error = new Error("It is not a number.");
            } else if (Number(text) < 0 || Number(text) > 1000) {
                error = new Error("The number must be between 0 and 1000.");
            }

            if (error) {
                error.name = "ValidationError";
                throw error;
            }
            return Number(text);
        }

        function square(n) {
            return n * n;
        }

        document.querySelector("#go").addEventListener("click", function () {
            const input = document.querySelector("#n");
            const out = document.querySelector("#out");
            out.style.color = "";
            try {
                const n = readNumber(input);
                out.textContent = `${n} squared is ${square(n)}`;
            } catch (error) {
                out.textContent = error.message;
                out.style.color = "red";
            } finally {
                input.value = "";   // always empty, with or without error
            }
        });
    </script>
</body>

</html>
El finally s'executa sempre, tant si ha anat bé com si hi ha hagut un error: el camp queda buit en tots dos casos.

Exercici 9.4

Exercise 9.4. Call a function that does not exist inside a try. In the catch, show a different message for your own errors (error.name === "ValidationError") and for the browser's errors.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>Two kinds of error</title>
</head>

<body>
    <button id="own">Own error</button>
    <button id="browser">Browser error</button>
    <p id="out"></p>

    <script>
        const out = document.querySelector("#out");

        function show(action) {
            try {
                action();
            } catch (error) {
                if (error.name === "ValidationError") {
                    out.textContent = "My own error: " + error.message;
                } else {
                    out.textContent = "Browser error (" + error.name + "): " + error.message;
                }
            }
        }

        document.querySelector("#own").addEventListener("click", function () {
            show(function () {
                const error = new Error("The age cannot be negative.");
                error.name = "ValidationError";
                throw error;
            });
        });

        document.querySelector("#browser").addEventListener("click", function () {
            show(function () {
                thisFunctionDoesNotExist();   // ReferenceError
            });
        });
    </script>
</body>

</html>

Mini-projecte 1

Mini-project 1: My toolbox page

One page with at least eight small tools. Each tool is one input (or two), one button, and one place where the answer appears. Behind each button there is one function that returns a value.

Ideas: convert units, calculate a mark average, check if a word is a palindrome, count the words in a sentence, calculate a price with VAT and discount, give a random number between two values, say if a year is a leap year, draw a row of N stars, calculate how many days until a date.

Rules:

  • every calculation must be a function that uses return (not one that writes in the page);
  • one separate function reads the inputs and shows the answer;
  • every function needs a comment above it saying what it does and what it returns;
  • clear names in camelCase, 4-space indentation;
  • no errors in the console when you use the page normally;
  • every calculation function has at least two console.assert that pass;
  • at least one function throws an error when the data is not valid, and the page catches it and shows the message.

Solució

index.html 
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="utf-8">
    <title>My toolbox</title>
</head>

<body>
    <h1>My toolbox</h1>

    <h2>1. Kilometres to miles</h2>
    <input type="text" id="km" placeholder="Kilometres">
    <button id="kmButton">Convert</button>
    <p id="kmOut"></p>

    <h2>2. Average of marks</h2>
    <input type="text" id="marks" placeholder="e.g. 5, 7, 9">
    <button id="marksButton">Average</button>
    <p id="marksOut"></p>

    <h2>3. Palindrome</h2>
    <input type="text" id="word" placeholder="A word">
    <button id="wordButton">Check</button>
    <p id="wordOut"></p>

    <h2>4. Count words</h2>
    <input type="text" id="sentence" placeholder="A sentence">
    <button id="sentenceButton">Count</button>
    <p id="sentenceOut"></p>

    <h2>5. Price with VAT and discount</h2>
    <input type="text" id="price" placeholder="Price">
    <input type="text" id="discount" placeholder="Discount %">
    <button id="priceButton">Calculate</button>
    <p id="priceOut"></p>

    <h2>6. Random number</h2>
    <input type="text" id="min" placeholder="Minimum">
    <input type="text" id="max" placeholder="Maximum">
    <button id="randomButton">Draw</button>
    <p id="randomOut"></p>

    <h2>7. Leap year</h2>
    <input type="text" id="year" placeholder="A year">
    <button id="yearButton">Check</button>
    <p id="yearOut"></p>

    <h2>8. Row of stars</h2>
    <input type="text" id="stars" placeholder="How many?">
    <button id="starsButton">Draw</button>
    <p id="starsOut"></p>

    <script>
        /* ---------- Calculation functions: they only return a value ---------- */

        /** Converts kilometres to miles. Returns a number. */
        function kmToMiles(km) {
            return km * 0.621371;
        }

        /** Average of an array of numbers. Returns a number. */
        function average(list) {
            let sum = 0;
            for (const n of list) {
                sum += n;
            }
            return sum / list.length;
        }

        /** True if the word reads the same backwards. Returns a boolean. */
        function isPalindrome(word) {
            const clean = word.toLowerCase();
            return clean === clean.split("").reverse().join("");
        }

        /** Counts the words of a sentence. Returns a number. */
        function countWords(sentence) {
            const text = sentence.trim();
            return text === "" ? 0 : text.split(/\s+/).length;
        }

        /** Price with a discount (in %) and 21 % VAT. Returns a number. */
        function finalPrice(price, discount) {
            return price * (1 - discount / 100) * 1.21;
        }

        /** Random whole number between min and max (both included). Throws if min > max. */
        function randomBetween(min, max) {
            if (min > max) {
                throw new Error("The minimum cannot be bigger than the maximum.");
            }
            return Math.floor(Math.random() * (max - min + 1)) + min;
        }

        /** True if the year is a leap year. Returns a boolean. */
        function isLeapYear(year) {
            return (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0;
        }

        /** A row with n stars. Throws if n is not between 1 and 50. */
        function starsRow(n) {
            if (n < 1 || n > 50) {
                throw new Error("Choose a number between 1 and 50.");
            }
            return "*".repeat(n);
        }

        /* ---------- Reading and validation ---------- */

        /** Reads a number from a field. Throws a ValidationError if it is not valid. */
        function readNumber(id) {
            const text = document.querySelector("#" + id).value.trim().replace(",", ".");
            if (text === "" || isNaN(Number(text))) {
                const error = new Error("Type a valid number.");
                error.name = "ValidationError";
                throw error;
            }
            return Number(text);
        }

        /** Runs one tool: calls the function and shows the answer (or the error). */
        function runTool(buttonId, outId, tool) {
            document.querySelector("#" + buttonId).addEventListener("click", function () {
                const out = document.querySelector("#" + outId);
                out.style.color = "";
                try {
                    out.textContent = tool();
                } catch (error) {
                    out.textContent = error.message;
                    out.style.color = "red";
                }
            });
        }

        runTool("kmButton", "kmOut", function () {
            return readNumber("km") + " km = " + kmToMiles(readNumber("km")).toFixed(2) + " miles";
        });

        runTool("marksButton", "marksOut", function () {
            const list = document.querySelector("#marks").value.split(",").map(Number);
            if (list.some(isNaN)) {
                throw new Error("Type numbers separated by commas.");
            }
            return "Average: " + average(list).toFixed(2);
        });

        runTool("wordButton", "wordOut", function () {
            const word = document.querySelector("#word").value;
            return isPalindrome(word) ? "Yes, it is a palindrome." : "No, it is not.";
        });

        runTool("sentenceButton", "sentenceOut", function () {
            return "Words: " + countWords(document.querySelector("#sentence").value);
        });

        runTool("priceButton", "priceOut", function () {
            return "Final price: " + finalPrice(readNumber("price"), readNumber("discount")).toFixed(2) + " EUR";
        });

        runTool("randomButton", "randomOut", function () {
            return "Your number: " + randomBetween(readNumber("min"), readNumber("max"));
        });

        runTool("yearButton", "yearOut", function () {
            return isLeapYear(readNumber("year")) ? "Leap year." : "Not a leap year.";
        });

        runTool("starsButton", "starsOut", function () {
            return starsRow(readNumber("stars"));
        });

        /* ---------- Tests: at least two console.assert per function ---------- */
        console.assert(Math.abs(kmToMiles(10) - 6.21371) < 0.0001, "kmToMiles 10");
        console.assert(kmToMiles(0) === 0, "kmToMiles 0");
        console.assert(average([5, 7, 9]) === 7, "average 5,7,9");
        console.assert(average([10]) === 10, "average of one");
        console.assert(isPalindrome("Level") === true, "palindrome Level");
        console.assert(isPalindrome("house") === false, "palindrome house");
        console.assert(countWords("one two  three") === 3, "countWords 3");
        console.assert(countWords("   ") === 0, "countWords empty");
        console.assert(Math.abs(finalPrice(100, 0) - 121) < 0.001, "finalPrice no discount");
        console.assert(Math.abs(finalPrice(100, 50) - 60.5) < 0.001, "finalPrice 50 %");
        console.assert(randomBetween(3, 3) === 3, "random 3,3");
        console.assert(randomBetween(1, 6) >= 1 && randomBetween(1, 6) <= 6, "random 1-6");
        console.assert(isLeapYear(2024) === true, "leap 2024");
        console.assert(isLeapYear(1900) === false, "leap 1900");
        console.assert(starsRow(3) === "***", "stars 3");
        console.assert(starsRow(1) === "*", "stars 1");
    </script>
</body>

</html>
Vuit eines; cada càlcul és una funció amb return i un comentari; readNumber llança un ValidationError i runTool el captura i el mostra en roig; cada funció té almenys dos console.assert que passen (la consola queda sense errors).
